\subsection*{Exercise~\ref{ex:sie_mass}: Enclosed Mass for an SIE Lens}

\textbf{(a)} For a flat $\Lambda$CDM cosmology with $\Hubble = 70$~km/s/Mpc
and $\Omegam = 0.3$, the angular diameter distances are computed via:
\begin{equation*}
    D(z_1, z_2) = \frac{\speedoflight}{(1+z_2)\,\Hubble}
    \int_{z_1}^{z_2} \frac{dz'}{E(z')}
\end{equation*}
where $E(z) = \sqrt{\Omegam(1+z)^3 + \OmegaL}$.  Numerical
evaluation gives:
\begin{align*}
    \Dd &\approx 919~\mathrm{Mpc} \,, \\
    \Ds &\approx 1652~\mathrm{Mpc} \,, \\
    \Dds &\approx 1055~\mathrm{Mpc} \,.
\end{align*}
(These match \texttt{problems\_09.wl}: 918.8, 1652.0 and 1054.7~Mpc.)

\textbf{(b)} The critical surface mass density is:
\begin{equation*}
    \Sigmacr = \frac{\speedoflight^2}{4\pi \Gnewton}
    \,\frac{\Ds}{\Dd\,\Dds}
    \approx 2.8 \times 10^9\;\Msun\,\mathrm{kpc}^{-2} \,.
\end{equation*}

\textbf{(c)} The enclosed projected mass is:
\begin{equation*}
    M_{\mathrm{E}} = \pi\,\thetaE^2\,\Dd^2\,\Sigmacr
    = \frac{\speedoflight^2}{4\Gnewton}\,\thetaE^2\,
    \frac{\Dd\,\Ds}{\Dds} \,.
\end{equation*}
Converting $\thetaE = 1.5'' = 7.27 \times 10^{-6}$~rad:
\begin{equation*}
    M_{\mathrm{E}} \approx 4.0 \times 10^{11}\;\Msun \,.
\end{equation*}
This is larger than the typical stellar mass of a massive elliptical
($\sim 10^{11}\;\Msun$), indicating significant dark matter within
the Einstein radius. \checkmark

\begin{derivation}[The arithmetic in physical units]
Work in kpc: $\RE = \Dd\thetaE = 918.8~\mathrm{Mpc}\times 7.27\times10^{-6} =
6.68$~kpc.  Then $M_{\mathrm{E}} = \pi\RE^2\Sigmacr = \pi(6.68)^2\times
2.83\times10^{9} = 3.98\times10^{11}\,\Msun$.  The axis ratio $q = 0.8$ never
enters: for the SIE, the mean convergence inside the critical ellipse is
exactly 1 and its area is $\pi\thetaE^2$ (chapter,
eq.~\eqref{eq:mass_from_thetaE} and the annotation that follows it).
\end{derivation}

\textbf{(d)} From $\thetaE = 4\pi\sigma_v^2\Dds/(\speedoflight^2\Ds)$:
\begin{equation*}
    \sigma_v = \speedoflight\,\sqrt{\frac{\thetaE\,\Ds}{4\pi\,\Dds}}
    \approx 285~\mathrm{km/s} \,.
\end{equation*}
This is typical of a massive early-type galaxy. \checkmark

\begin{derivation}[Inverting for $\sigma_v$]
$\thetaE = 4\pi(\sigma_v/\speedoflight)^2\,\Dds/\Ds \Rightarrow
(\sigma_v/\speedoflight)^2 = \thetaE\Ds/(4\pi\Dds)$, with $\thetaE$ in radians.
Here $\Ds/\Dds = 1652.0/1054.7 = 1.566$, so
$\sigma_v = \speedoflight\sqrt{7.27\times10^{-6}\times1.566/(4\pi)} = 285$~km/s.
Unit trap: using $\thetaE$ in arcsec inflates $\sigma_v$ by
$\sqrt{206265} \approx 454$.
\end{derivation}

\subsection*{Exercise~\ref{ex:msd_derivation}: Mass-Sheet Degeneracy}

\textbf{(a)} The original deflection satisfies
$\nabla \cdot \vecalpha = 2\kappab$.  Under the MST:
\begin{equation*}
    \kappab_\lambda = \lambda\kappab + (1-\lambda)
    \implies
    \vecalpha_\lambda = \lambda\vecalpha + (1-\lambda)\vectheta \,.
\end{equation*}
The transformed lens equation:
\begin{align*}
    \vecbeta_\lambda
    &= \vectheta - \vecalpha_\lambda(\vectheta) \\
    &= \vectheta - \lambda\vecalpha(\vectheta)
       - (1-\lambda)\vectheta \\
    &= \lambda[\vectheta - \vecalpha(\vectheta)] \\
    &= \lambda\vecbeta \,.
\end{align*}
So the same image positions $\vectheta$ satisfy the transformed lens
equation for source position $\lambda\vecbeta$. \checkmark

\begin{annotation}[The step ``$\Rightarrow$'' in the first display]
$\vecalpha_\lambda$ follows from $\kappab_\lambda$ only up to a curl-free,
divergence-free field.  The MST is the choice $\psiL_\lambda = \lambda\psiL +
\tfrac12(1-\lambda)|\vectheta|^2$, whose gradient is
$\lambda\vecalpha + (1-\lambda)\vectheta$.  The sheet term works because a
uniform convergence $\kappab_s$ deflects by $\kappab_s\vectheta$:
$\nabla\cdot(\kappab_s\vectheta) = 2\kappab_s$ in two dimensions.
\end{annotation}


\textbf{(b)} The Jacobian is
$\Amat = \mathbb{I} - \partial\vecalpha/\partial\vectheta$.
Under the MST:
\begin{align*}
    \Amat_\lambda
    &= \mathbb{I}
       - \frac{\partial\vecalpha_\lambda}{\partial\vectheta} \\
    &= \mathbb{I} - \lambda\frac{\partial\vecalpha}{\partial\vectheta}
       - (1-\lambda)\mathbb{I} \\
    &= \lambda\left(\mathbb{I}
       - \frac{\partial\vecalpha}{\partial\vectheta}\right)
    = \lambda\,\Amat \,.
\end{align*}
Therefore $\det\Amat_\lambda = \lambda^2\det\Amat$, and
$\mu_\lambda = 1/\det\Amat_\lambda = \mu/\lambda^2$. \checkmark

\textbf{(c)} Since $\mu_{\lambda,i} = \mu_i/\lambda^2$ for every
image $i$, the ratio:
\begin{equation*}
    \frac{\mu_{\lambda,i}}{\mu_{\lambda,j}}
    = \frac{\mu_i/\lambda^2}{\mu_j/\lambda^2}
    = \frac{\mu_i}{\mu_j} \,.
\end{equation*}
The $\lambda^2$ factors cancel.  Flux ratios are invariant. \checkmark

\textbf{(d)} The Fermat potential is
$\phi(\vectheta, \vecbeta) = \frac{1}{2}|\vectheta - \vecbeta|^2
- \psiL(\vectheta)$.  Under the MST, the lensing potential
transforms as $\psiL_\lambda = \lambda\psiL +
\frac{1}{2}(1-\lambda)|\vectheta|^2$.  The time delay between
images $A$ and $B$ is:
\begin{align*}
    \Delta t_\lambda
    &= \frac{(1+z_d)\Dd\Ds}{\speedoflight\,\Dds}
       [\phi_\lambda(\vectheta_A, \vecbeta_\lambda)
        - \phi_\lambda(\vectheta_B, \vecbeta_\lambda)] \\
    &= \lambda\,\Delta t \,.
\end{align*}
The derivation proceeds by substituting $\vecbeta_\lambda =
\lambda\vecbeta$ and $\psiL_\lambda = \lambda\psiL +
\frac{1}{2}(1-\lambda)|\vectheta|^2$ and simplifying.  The
uniform $|\vectheta|^2$ term cancels between the two images
because they are at the same image positions, leaving a factor
of $\lambda$ multiplying the original Fermat potential
difference. \checkmark

\begin{derivation}[The simplification, every line]
Expand $\phi_\lambda(\vectheta,\lambda\vecbeta)$:
\begin{align*}
  \phi_\lambda
  &= \tfrac12|\vectheta|^2 - \lambda\,\vectheta\cdot\vecbeta + \tfrac12\lambda^2|\vecbeta|^2
     - \lambda\psiL - \tfrac12|\vectheta|^2 + \tfrac12\lambda|\vectheta|^2\\
  &= \lambda\big[\tfrac12|\vectheta|^2 - \vectheta\cdot\vecbeta - \psiL\big] + \tfrac12\lambda^2|\vecbeta|^2\\
  &= \lambda\big[\tfrac12|\vectheta-\vecbeta|^2 - \psiL\big] - \tfrac12\lambda|\vecbeta|^2 + \tfrac12\lambda^2|\vecbeta|^2
   = \lambda\,\phi(\vectheta,\vecbeta) + \tfrac12\lambda(\lambda-1)|\vecbeta|^2 .
\end{align*}
The $|\vectheta|^2$ pieces cancel \emph{within} each image (line~1 to
line~2).  What cancels \emph{between} images is the last term,
$\tfrac12\lambda(\lambda-1)|\vecbeta|^2$, which depends only on the common
source.  Hence $\Delta\phi_\lambda = \lambda\Delta\phi$ and, at fixed
distances, $\Delta t_\lambda = \lambda\Delta t$.
\end{derivation}

\begin{annotation}[Checklist: what an MST leaves alone]
Invariant: image positions, flux ratios, critical curves, the Einstein radius,
and the reduced shear $\gammaL/(1-\kappab)$.  Changed: $\kappab$, $\gammaL$,
$|\mu|$ ($\lambda^{-2}$), $\Delta t$ ($\lambda$), source size and position
($\lambda$), caustics ($\lambda$).  Any observable in the first list cannot
break the degeneracy; one in the second list can, if its absolute value is
known independently (a standard candle for $|\mu|$, a cosmology for
$\Delta t$, or a dynamical mass).
\end{annotation}

\subsection*{Exercise~\ref{ex:H0_time_delay}: $\Hubble$ from a Time-Delay Lens}

\textbf{(a)} The time-delay equation is:
\begin{equation*}
    \Delta t = \frac{(1+z_d)}{\speedoflight}
    \,\frac{\Dd\,\Ds}{\Dds}\,\Delta\tau \,.
\end{equation*}
Solving for $\Hubble$: since $\Dd\Ds/\Dds \propto 1/\Hubble$,
we can write
$\Dd\Ds/\Dds = d(z_d, z_s, \Omegam)/\Hubble$ where $d$ is a
dimensionless function of redshifts and cosmological parameters.
Then:
\begin{equation*}
    \Hubble = \frac{(1+z_d)\,\Delta\tau}{\speedoflight\,\Delta t}
    \times d(z_d, z_s, \Omegam) \,.
\end{equation*}

\begin{annotation}[Convention check]
Here, as in the exercise hint, $(1+z_d)$ sits outside $d$, and
$d = \Hubble\Dd\Ds/\Dds$ is in km\,s$^{-1}$ (it equals $\speedoflight\,f_df_s/f_{ds}$
in the notation of the chapter's $D_{\Delta t}$ box).  The chapter's
eq.~\eqref{eq:time_delay_recap} calls $\Dd\Ds/\Dds$ alone $D_{\Delta t}$; the
script's printout calls $(1+z_d)\Dd\Ds/\Dds$ by that name.  Both conventions
are in use.  The $\Hubble$ you get is the same provided $(1+z_d)$ appears
exactly once.
\end{annotation}


\textbf{(b)} For $z_d = 0.6$, $z_s = 1.5$, $\Omegam = 0.3$
(flat $\Lambda$CDM), the dimensionless distance ratio
$d = \Hubble\Dd\Ds/\Dds$ is computed from the cosmological
integrals.  Converting $\Delta\tau = 0.64~\mathrm{arcsec}^2
= 0.64 \times (4.848 \times 10^{-6})^2~\mathrm{rad}^2
= 1.50 \times 10^{-11}~\mathrm{rad}^2$ and
$\Delta t = 80~\mathrm{days} = 6.91 \times 10^6~\mathrm{s}$:

Numerical evaluation gives $\Hubble \approx 70$~km/s/Mpc
(the exact value depends on the precise cosmological distances;
see \texttt{problems\_09.wl} for the detailed calculation). \checkmark

\begin{derivation}[The ``self-consistent'' step without iteration]
Because $D_{\Delta t} \propto 1/\Hubble$ exactly, no iteration is needed.
Evaluate at a reference $H_{\mathrm{ref}} = 70$:
$(1+z_d)\Dd\Ds/\Dds = 1.6\times1379.0\times1745.6/863.0 = 4463$~Mpc.  The
predicted delay is
$\Delta t_{\mathrm{pred}} = D_{\Delta t}\Delta\tau/\speedoflight =
(4463\times3.086\times10^{24}~\mathrm{cm})(1.504\times10^{-11})/
(2.998\times10^{10}~\mathrm{cm\,s^{-1}}) = 79.98$~days.  Then
\[
  \Hubble = H_{\mathrm{ref}}\,\frac{\Delta t_{\mathrm{pred}}(H_{\mathrm{ref}})}{\Delta t_{\mathrm{obs}}}
  = 70\times\frac{79.98}{80} = 69.98~\mathrm{km\,s^{-1}\,Mpc^{-1}} .
\]
\end{derivation}


\textbf{(c)} If the true lens has an unmodeled positive mass sheet
with $\kappab_s = 0.05$, then the true MST parameter is
$\lambda = 1 - \kappab_s = 0.95$.  Recall (Section~9.2,
eq.~\ref{eq:mst_H0}) that under the MST the model-predicted time
delay scales as $\Delta t_\lambda = \lambda\,\Delta t$ at fixed
$\Hubble$, so at \emph{fixed observed} $\Delta t$ the inferred
$\Hubble$ scales the opposite way:
\begin{equation*}
    \Hubble^{\mathrm{true}} = \lambda\,\Hubble^{\mathrm{naive}}
    \quad\Longleftrightarrow\quad
    \Hubble^{\mathrm{naive}} = \Hubble^{\mathrm{true}} / \lambda \,.
\end{equation*}
A naive ($\lambda = 1$) analysis therefore \emph{overestimates}
$\Hubble$ by $1/\lambda - 1 \approx 5\%$ when the lens actually has a
$5\%$ positive mass sheet.  For
$\Hubble^{\mathrm{naive}} \approx 73$~km/s/Mpc, this gives
$\Hubble^{\mathrm{true}} \approx 69$~km/s/Mpc --- closer to the Planck
CMB value.  This is the direction in which TDCOSMO has pulled the
time-delay $\Hubble$ relative to the original H0LiCOW result, and it
sits squarely in the modern $\Hubble$-tension discussion. \checkmark

\begin{derivation}[The sign, in one chain]
$\Delta t_{\mathrm{obs}} = D_{\Delta t}(\Hubble)\,\Delta\tau/\speedoflight$ with
$D_{\Delta t} \propto \Hubble^{-1}$, so $\Hubble \propto \Delta\tau$ at fixed
$\Delta t_{\mathrm{obs}}$.  The true model is the MST of the fitted one, with
$\Delta\tau_{\mathrm{true}} = \lambda\Delta\tau_{\mathrm{naive}}$, hence
$\Hubble^{\mathrm{true}} = \lambda\Hubble^{\mathrm{naive}}$.  With this
exercise's own number, $0.95\times69.98 = 66.48$; with the round $73$ of the
text, $0.95\times73 = 69.35 \approx 69$.
\end{derivation}

\subsection*{Exercise~\ref{ex:power_law_profile}: Power-Law Profile}

\textbf{(a)} For $\rho(r) = \rho_0(r/r_0)^{-\gamma'}$, the projected
surface mass density is:
\begin{equation*}
    \Sigma(R) = \int_{-\infty}^{\infty} \rho(\sqrt{R^2 + z^2})\,dz
    = \rho_0\,r_0^{\gamma'}\,\frac{\sqrt{\pi}\,\Gamma[(\gamma'-1)/2]}
    {\Gamma[\gamma'/2]}\,R^{1-\gamma'}
\end{equation*}
where $R = \Dd\theta$ is the projected physical radius.  The
convergence is $\kappab = \Sigma/\Sigmacr$, so:
\begin{equation*}
    \kappab(\theta) \propto \theta^{1-\gamma'} \,.
\end{equation*}
More precisely, $\kappab(\theta) = \frac{3-\gamma'}{2}
(\thetaE/\theta)^{\gamma'-1}$, where $\thetaE$ is defined as the
radius where the mean convergence equals unity. \checkmark

\begin{derivation}[From $\rho$ to the normalized $\kappab$, all steps]
\begin{enumerate}[nosep]
  \item $\Sigma(R) = \rho_0r_0^{\gamma'}\int_{-\infty}^{\infty}(R^2+z^2)^{-\gamma'/2}dz$.
        Put $z = Ru$:
        $= \rho_0r_0^{\gamma'}R^{1-\gamma'}\cdot2\int_0^\infty(1+u^2)^{-\gamma'/2}du$.
  \item Put $t = u^2/(1+u^2)$: $2\int_0^\infty(1+u^2)^{-\gamma'/2}du =
        \int_0^1t^{-1/2}(1-t)^{(\gamma'-3)/2}dt = B\big(\tfrac12,\tfrac{\gamma'-1}{2}\big)
        = \sqrt\pi\,\Gamma\big(\tfrac{\gamma'-1}{2}\big)/\Gamma\big(\tfrac{\gamma'}{2}\big)$
        (convergent for $\gamma' > 1$).  This is the prefactor above.
  \item $\kappab = \Sigma/\Sigmacr \propto \theta^{1-\gamma'}$; write
        $\kappab = A(\thetaE/\theta)^{\gamma'-1}$.
  \item $\bar\kappab(\theta) = \frac{2}{\theta^2}\int_0^\theta\kappab\theta'd\theta'
        = \frac{2A}{3-\gamma'}(\thetaE/\theta)^{\gamma'-1}$ for $\gamma' < 3$.
  \item $\bar\kappab(\thetaE) = 1 \Rightarrow A = (3-\gamma')/2$.
\end{enumerate}
The window $1 < \gamma' < 3$ is where both the line-of-sight integral and the
enclosed 2D mass are finite.
\end{derivation}


\textbf{(b)} For $\gamma' = 2$:
$\kappab \propto \theta^{1-2} = \theta^{-1} = 1/\theta$, which
is the SIS convergence $\kappab = \thetaE/(2\theta)$. \checkmark

\textbf{(c)} The enclosed mass within radius $R$ scales as
$M(\le R) \propto R^{3-\gamma'}$.  The Einstein mass is
$M_{\mathrm{E}} = M(\le R_{\mathrm{E}})$, fixed by lensing.
The velocity dispersion scales as:
\begin{equation*}
    \sigma_v^2 \propto \frac{M(\le R_{\mathrm{ap}})}{R_{\mathrm{ap}}}
    \propto R_{\mathrm{ap}}^{2-\gamma'} \,.
\end{equation*}
Since $R_{\mathrm{ap}} \neq R_{\mathrm{E}}$ in general, the ratio
$\sigma_v^2/M_{\mathrm{E}}$ depends on $\gamma'$, breaking the
degeneracy between mass normalization and profile slope. \checkmark

\begin{derivation}[Where $\sigma_v^2 \propto M/R$ comes from]
For an isotropic tracer $\nu \propto r^{-\delta}$ in $M(r) \propto
r^{3-\gamma'}$, integrating the spherical Jeans equation
$d(\nu\sigma_r^2)/dr = -\nu\Gnewton M/r^2$ inward from infinity gives
\[
  \sigma_r^2 = \frac{\Gnewton M(r)}{r}\,\frac{1}{\gamma'+\delta-2}
  \qquad(\gamma'+\delta > 2),
\]
and $1/[2(\gamma'-1)]$ if the light traces the mass ($\delta = \gamma'$).  The
$\gamma'$-dependent factor is part of $g(\gamma')$; projection and aperture
averaging add more.  With $M_{\mathrm{E}}$ fixed at $\RE$ and
$M_{\mathrm{3D}}(r) \propto r^{3-\gamma'}$, the prediction at $R_{\mathrm{ap}}$
carries $(R_{\mathrm{ap}}/\RE)^{2-\gamma'}$.  This is the extra $\gamma'$
dependence that lensing alone does not have.
\end{derivation}


\textbf{(d)} For a given $\thetaE$ and $\sigma_v$, the self-consistent
$\gamma'$ can be found from the relation linking both.  For
$\thetaE = 1.2''$ and $\sigma_v = 230$~km/s, numerical evaluation
(see \texttt{problems\_09.wl}) gives $\gamma' \approx 2.0$--$2.1$,
fully consistent with the bulge-halo conspiracy value
$\gamma' \approx 2.0 \pm 0.2$. \checkmark

\begin{annotation}[What the script actually computes]
The exercise gives no redshifts.  \texttt{problems\_09.wl} assumes
$z_d = 0.3$, $z_s = 1.0$ and computes the isothermal prediction for
$\thetaE = 1.2''$: $\sigma_{\mathrm{SIS}} = 255$~km/s, so
$\sigma_{\mathrm{obs}}/\sigma_{\mathrm{SIS}} = 0.90$ and
$(\sigma_{\mathrm{obs}}/\sigma_{\mathrm{SIS}})^2 = 0.81$.  It stops there.  It
does not invert $g(\gamma')$, because that requires an aperture, a light
profile and an anisotropy that the exercise does not specify.  Treat the
quoted $\gamma' \approx 2.0$--$2.1$ as a statement that the ratio is of order
unity, not as a computed value.  For a real inversion, Part~V
(Chapter~\ref{ch:msd_kinematics}) sets up the full projected,
aperture-weighted Jeans prediction.
\end{annotation}

\textbf{(d)} B1608+656 has both the largest enclosed mass ($\sim 3.9 \times
10^{11}\,\Msun$) and the largest physical Einstein radius
($\sim 6.8$~kpc), despite having a similar angular Einstein
radius to Q0957+561.  This is because B1608+656 is at higher
redshift ($z_d = 0.63$), so the same angular size corresponds to
a larger physical size and a larger mass.  Q2237+0305 has the
smallest mass and physical radius because its lens is so nearby
($z_d = 0.04$) --- the angular diameter distance $\Dd$ is small,
so the physical Einstein radius is only $\sim 0.7$~kpc even though
$\thetaE \approx 0.9''$. \checkmark
