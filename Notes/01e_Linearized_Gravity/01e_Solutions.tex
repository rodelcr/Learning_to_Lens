\subsection*{Exercise~\ref{ex:newtonian_limit_prob}: Newtonian Limit from the Geodesic Equation}

\textbf{(a)} The Christoffel symbols to first order in $h$ are
(eq.~\eqref{eq:christoffel_linearized}):
\[
    \christoffel{\alpha}{\mu}{\nu}
    = \half \eta^{\alpha\beta}\bigl(
    \partial_\mu h_{\nu\beta} + \partial_\nu h_{\mu\beta}
    - \partial_\beta h_{\mu\nu}\bigr) \,.
\]
For $\mu = \nu = 0$ and $\alpha = i$ (spatial):
\[
    \christoffel{i}{0}{0}
    = \half \eta^{i\beta}\bigl(
    \partial_0 h_{0\beta} + \partial_0 h_{0\beta}
    - \partial_\beta h_{00}\bigr) \,.
\]
Using the static-field condition $\partial_0 h_{\mu\nu} = 0$,
the first two terms vanish.  Since $\eta^{ij} = \delta^{ij}$ and
$\eta^{i0} = 0$:
\[
    \boxed{\christoffel{i}{0}{0} = -\half \partial_i h_{00}}
\]

\begin{derivation}[The index sum written out]
$\eta^{i\beta}$ is nonzero only for $\beta = j$ spatial, where
$\eta^{ij} = \delta^{ij}$ ($\eta^{i0} = 0$):
\begin{align*}
\christoffel{i}{0}{0}
&= \tfrac12\sum_{\beta}\eta^{i\beta}\bigl(2\,\partial_0 h_{0\beta} - \partial_\beta h_{00}\bigr) \\
&= \tfrac12\sum_{j}\delta^{ij}\bigl(2\,\partial_0 h_{0j} - \partial_j h_{00}\bigr)
  && (\eta^{i0} = 0) \\
&= \tfrac12\bigl(0 - \partial_i h_{00}\bigr)
  && (\text{static: } \partial_0 h_{0j} = 0;\ \delta^{ij} \text{ picks } j = i) \\
&= -\tfrac12\,\partial_i h_{00} .
\end{align*}
For the weak-field metric, where $h_{00} = -2\Phi/c^2$, this is
$+\partial_i\Phi/c^2$, which points away from the mass.  The minus sign in
the geodesic equation then turns it into an attraction.
\end{derivation}


\textbf{(b)} The geodesic equation for spatial components is:
\[
    \frac{d^2 x^i}{d\tau^2}
    + \christoffel{i}{\alpha}{\beta}\,
    \frac{dx^\alpha}{d\tau}\, \frac{dx^\beta}{d\tau} = 0 \,.
\]
In the slow-motion limit, $|dx^j/d\tau| \ll c\,dt/d\tau$, so we keep
only the $\alpha = \beta = 0$ term:
\[
    \frac{d^2 x^i}{d\tau^2}
    + \christoffel{i}{0}{0}\left(\frac{dx^0}{d\tau}\right)^{\!2} = 0 \,.
\]
Substituting $\christoffel{i}{0}{0} = -\half \partial_i h_{00}$ and
$dx^0/d\tau = c\, dt/d\tau$:
\[
    \frac{d^2 x^i}{d\tau^2}
    = \half c^2 \partial_i h_{00}\left(\frac{dt}{d\tau}\right)^{\!2} .
\]
Since $d\tau \approx dt$ for slow motion ($\gamma \approx 1$):
\[
    \boxed{\frac{d^2 x^i}{dt^2} = \half c^2\, \partial_i h_{00}}
\]

\begin{annotation}[What was dropped, and why it is safe]
\begin{itemize}
    \item $\christoffel{i}{0}{j}$ terms: zero for a static field with
          $h_{0i} = 0$.
    \item $\christoffel{i}{j}{k}\,(dx^j/d\tau)(dx^k/d\tau)$: of size
          $|\nabla\Phi|\,v^2/c^2$, i.e.\ smaller than the kept term by
          $v^2/c^2$ (about $10^{-8}$ for the Earth's orbital speed).
    \item $d\tau \approx dt$: not actually needed.  Since
          $d^2t/d\tau^2$ is suppressed by $v/c$, $dt/d\tau$ is constant to
          this order, and the factor $(dt/d\tau)^2$ appears on both sides
          and cancels.
\end{itemize}
\end{annotation}


\textbf{(c)} Comparing with Newton's law $d^2 x^i/dt^2 = -\partial_i \Phi$:
\[
    -\partial_i \Phi = \half c^2\, \partial_i h_{00}
    \quad\Longrightarrow\quad
    \boxed{h_{00} = -\frac{2\Phi}{c^2}}
\]

\begin{derivation}[The integration constant]
Equal gradients only give $\partial_i\bigl(h_{00} + 2\Phi/c^2\bigr) = 0$, so
$h_{00} = -2\Phi/c^2 + C$.  Far from all sources the metric is Minkowski
($h_{00} \to 0$) and the Newtonian potential is normalised to vanish
($\Phi \to 0$), hence $C = 0$.  For a point mass:
$h_{00} = 2GM/(c^2r) = R_S/r$.
\end{derivation}

\subsection*{Exercise~\ref{ex:weak_field_schwarzschild}: Weak-Field Schwarzschild}

\textbf{(a)} The Schwarzschild metric has
$g_{rr} = (1 - R_S/r)^{-1}$.  Expanding to first order in
$\epsilon \equiv R_S/r \ll 1$:
\[
    (1 - \epsilon)^{-1} = 1 + \epsilon + \epsilon^2 + \cdots
    \approx 1 + \frac{R_S}{r} \,.
\]
So the metric becomes:
\[
    ds^2 \approx -\left(1 - \frac{R_S}{r}\right) c^2 dt^2
    + \left(1 + \frac{R_S}{r}\right) dr^2
    + r^2\, d\Omega^2 \,.
\]

\textbf{(b)} Relabelling the Schwarzschild $r$ as the Cartesian radius
does not work: $(1 + R_S/r)\,dr^2 + r^2 d\Omega^2$ and
$(1 + R_S/r)(dr^2 + r^2 d\Omega^2)$ differ by $R_S\,r\,d\Omega^2$, which is
\emph{first} order in $R_S/r$ relative to $r^2 d\Omega^2$.\footnote{Earlier
versions of this solution said this difference ``only matters at second
order in $R_S/r$''; it is first order, and the change of radius below is
what removes it.}  Instead change to the isotropic radius $\rho$,
$r = \rho\,(1 + R_S/4\rho)^2 \approx \rho + R_S/2$
(eq.~\eqref{eq:isotropic_r}).  To first order in $R_S/\rho$,
\[
    1 - \frac{R_S}{r} = 1 - \frac{R_S}{\rho} \,, \qquad
    \Bigl(1 + \frac{R_S}{r}\Bigr)dr^2 + r^2 d\Omega^2
    = \Bigl(1 + \frac{R_S}{\rho}\Bigr)\bigl(d\rho^2 + \rho^2 d\Omega^2\bigr) \,,
\]
and $d\rho^2 + \rho^2 d\Omega^2 = dx^2 + dy^2 + dz^2$ with
$\rho = |\mathbf{x}|$:
\[
    ds^2 \approx -\left(1 - \frac{R_S}{\rho}\right) c^2 dt^2
    + \left(1 + \frac{R_S}{\rho}\right)(dx^2 + dy^2 + dz^2) \,.
\]
With $\Phi = -GM/\rho$ (equal to $-GM/r$ up to terms of order
$(R_S/\rho)^2$ in $\Phi/c^2$) and $R_S = 2GM/c^2$:
\[
    1 - \frac{R_S}{\rho} = 1 + \frac{2\Phi}{c^2} \,, \qquad
    1 + \frac{R_S}{\rho} = 1 - \frac{2\Phi}{c^2} \,.
\]
This is exactly the weak-field metric~\eqref{eq:weak_field_metric}.

\begin{annotation}[Why the radius must change in (b)]
The missing $R_S\,r\,d\Omega^2$ is the same order as the terms being kept,
so no relabelling of $r$ can supply it.  The shift $r \simeq \rho + R_S/2$
does, through $r^2 \simeq \rho^2 + R_S\rho$.  The derivation box below
expands each factor.
\end{annotation}

\begin{derivation}[Part (b) in detail: the isotropic substitution]
Use $r = \rho\,(1 + R_S/4\rho)^2$ (eq.~\eqref{eq:isotropic_r}).  Expanding,
\begin{align*}
r &= \rho + \frac{R_S}{2} + \frac{R_S^2}{16\rho},
\qquad
\frac{dr}{d\rho} = 1 - \frac{R_S^2}{16\rho^2}
  && \text{(exact)} \\
1 - \frac{R_S}{r} &= 1 - \frac{R_S}{\rho} + \order{R_S^2/\rho^2}
  && (r = \rho + \order{R_S}) \\
\Bigl(1 + \frac{R_S}{r}\Bigr)dr^2 &= \Bigl(1 + \frac{R_S}{\rho}\Bigr)d\rho^2
  + \order{R_S^2/\rho^2}
  && (dr/d\rho = 1 + \order{R_S^2}) \\
r^2\,d\Omega^2 &= \rho^2\Bigl(1 + \frac{R_S}{\rho}\Bigr)d\Omega^2
  + \order{R_S^2/\rho^2}
  && \bigl(r^2 = \rho^2 + R_S\rho + \dots\bigr) .
\end{align*}
The shift $r = \rho + R_S/2$ supplies exactly the missing $R_S\,\rho\,d\Omega^2$:
the radial and angular parts acquire the \emph{same} factor
$1 + R_S/\rho$, so
\[
ds^2 \simeq -\Bigl(1 - \frac{R_S}{\rho}\Bigr)c^2dt^2
+ \Bigl(1 + \frac{R_S}{\rho}\Bigr)\bigl(d\rho^2 + \rho^2d\Omega^2\bigr) ,
\]
and $d\rho^2 + \rho^2d\Omega^2 = dx^2 + dy^2 + dz^2$ with $\rho = |\mathbf{x}|$.
The exact isotropic metric (eq.~\eqref{eq:schwarzschild_isotropic}) follows
from the same substitution without expanding:
$1 - R_S/r = \bigl[(1 - R_S/4\rho)/(1 + R_S/4\rho)\bigr]^2$,
$(dr/d\rho)^2/(1 - R_S/r) = (1 + R_S/4\rho)^4$ and
$r^2 = (1 + R_S/4\rho)^4\rho^2$.
\end{derivation}


\textbf{(c)} We identify $h_{00} = -2\Phi/c^2 = R_S/\rho$ and
$h_{ij} = -2\Phi/c^2\, \delta_{ij} = (R_S/\rho)\, \delta_{ij}$.
Thus $h_{00} = h_{xx} = h_{yy} = h_{zz} = R_S/\rho$ --- the time and
spatial diagonal perturbations are equal in magnitude.  This equality is
the origin of the factor-of-two in light bending.

\subsection*{Exercise~\ref{ex:refractive_index_prob}: Effective Refractive Index}

The effective refractive index is $n - 1 = 2GM/(c^2 r)$.
Using $G = 6.674 \times 10^{-11}$~m$^3$\,kg$^{-1}$\,s$^{-2}$,
$M_\odot = 1.989 \times 10^{30}$~kg, $c = 2.998 \times 10^8$~m/s:

\textbf{(a)} $r = R_\odot = 6.96 \times 10^8$~m:
\[
    n - 1 = \frac{2 \times 6.674\!\times\!10^{-11} \times 1.989\!\times\!10^{30}}
    {(2.998\!\times\!10^{8})^2 \times 6.96\!\times\!10^{8}}
    = \boxed{4.24 \times 10^{-6}}
\]
The weak-field approximation is excellent ($n - 1 \sim 10^{-6}$).

\textbf{(b)} $r = 1$~AU $= 1.496 \times 10^{11}$~m:
\[
    n - 1 = \frac{2GM_\odot}{c^2 \times 1.496\!\times\!10^{11}}
    = \boxed{1.97 \times 10^{-8}}
\]
Even weaker field; the Sun's effect at Earth's distance is tiny.

\textbf{(c)} $r = 1$~kpc $= 3.086 \times 10^{19}$~m:
\[
    n - 1 = \frac{2GM_\odot}{c^2 \times 3.086\!\times\!10^{19}}
    = \boxed{9.57 \times 10^{-17}}
\]
Completely negligible.  At kiloparsec scales, the gravitational
potential of a single solar mass is immeasurably weak; only the
cumulative effect of $\sim 10^{11}$--$10^{12}$ stars in a galaxy
produces observable lensing.

\begin{annotation}[A one-line mental check]
For a point mass $n - 1 = R_S/r$, and the Sun's Schwarzschild radius is
$R_S \approx 2.95$~km.  Then $n - 1 \approx 2.95/(6.96\times10^5) =
4.24\times10^{-6}$ at the limb, and the other two answers are this value
scaled by $R_\odot/r$.
\end{annotation}

\subsection*{Exercise~\ref{ex:deflection_angle_derivation}: Deflection Angle from Null Geodesics}

The unperturbed photon path is $\mathbf{x}(l) = (l, b, 0)$, where
$l$ is the coordinate along the path and $b$ is the impact parameter.

\textbf{(a)} The Newtonian potential is $\Phi = -GM/r = -GM/\sqrt{l^2 + b^2}$.
The perpendicular gradient (in the $y$-direction) is:
\[
    \nabla_\perp \Phi = \frac{\partial \Phi}{\partial b}
    = \frac{GM\, b}{(l^2 + b^2)^{3/2}} \,.
\]

\begin{annotation}[Signs in part (a)]
$\partial\Phi/\partial b = +GMb/(l^2 + b^2)^{3/2} > 0$: the gradient of
$\Phi$ points \emph{away} from the mass, because $\Phi = -GM/r$ increases
toward zero outward.  The gradient of $|\Phi| = GM/r$ is the negative of
this and points toward the mass, which is the direction of the bend.
Eq.~\eqref{eq:deflection_integral} needs only the magnitude, so both
give $GMb/(l^2 + b^2)^{3/2}$.  Part~(c) below uses $\nabla_\perp\Phi$ in
that sense.
\end{annotation}


\textbf{(b)} The integral is:
\[
    \int_{-\infty}^{\infty} \frac{b\, dl}{(l^2 + b^2)^{3/2}}
    = \frac{b}{b^2} \left[\frac{l}{\sqrt{l^2 + b^2}}\right]_{-\infty}^{+\infty}
    = \frac{1}{b}(1 - (-1)) = \frac{2}{b} \,.
\]

\begin{derivation}[Where the antiderivative comes from]
Check by differentiating with the quotient rule:
\[
\frac{d}{dl}\,\frac{l}{\sqrt{l^2 + b^2}}
= \frac{\sqrt{l^2 + b^2} - l\cdot l/\sqrt{l^2 + b^2}}{l^2 + b^2}
= \frac{(l^2 + b^2) - l^2}{(l^2 + b^2)^{3/2}}
= \frac{b^2}{(l^2 + b^2)^{3/2}} ,
\]
so $\int dl\,(l^2 + b^2)^{-3/2} = l/\bigl(b^2\sqrt{l^2 + b^2}\bigr)$, and the
prefactor $b$ gives the $b/b^2$ above.  The limits are
$l/\sqrt{l^2 + b^2} \to \pm1$ as $l \to \pm\infty$.  To discover the
antiderivative rather than verify it, substitute $l = b\tan\vartheta$;
the integrand becomes $\cos\vartheta\,d\vartheta/b$.
\end{derivation}


\textbf{(c)} The deflection integral
(eq.~\eqref{eq:deflection_integral}) uses $\nabla_\perp \Phi$ to denote
the gradient of $|\Phi|$ perpendicular to the path.  Its prefactor
$(1 + \gamma_{\mathrm{PPN}})/c^2 = 2/c^2$ already accounts for both the
time ($h_{00}$) and space ($h_{ij}$) contributions, each $1/c^2$; the
factor of~4 in $4GM/(c^2 b)$ is this~2 times the~2 from part~(b).  The
same result follows from the projected (2D) lens-plane formula.

Using the projected (2D) lens-plane formulation
(Schneider, Ehlers \& Falco 1992; Congdon \& Keeton eq.~3.78), the
deflection by a surface mass density $\Sigma$ is:
\[
    \boldsymbol{\hat{\alpha}} = \frac{4G}{c^2}
    \int \frac{(\vecxi - \vecxi')}{|\vecxi - \vecxi'|^2}\,
    \Sigma(\vecxi')\, d^2\xi' \,.
\]
For a point mass projected at the origin,
$\Sigma = M\,\delta^{(2)}(\vecxi')$, and the integral gives directly:
\[
    \boxed{\alphahat = \frac{4GM}{c^2 b}}
\]
This is verified symbolically in
\mathlink{01e_Linearized_Gravity/weak_field_metric.wl}{weak\_field\_metric.wl}.

Equivalently, using the PPN formalism (Weinberg 1972; Will 1993),
the deflection is:
\[
    \alphahat = \frac{1 + \gamma_{\mathrm{PPN}}}{c^2}
    \int_{-\infty}^{\infty} \nabla_\perp |\Phi|\, dl \,,
\]
where $\gamma_{\mathrm{PPN}} = 1$ in GR (the spatial curvature
contribution).  With $\int \nabla_\perp |\Phi|\, dl = 2GM/b$ and
$(1 + 1) = 2$, this gives $\alphahat = (2/c^2)(2GM/b) = 4GM/(c^2 b)$,
confirming the result.

\textbf{(d)} For the Sun at the solar limb:
\begin{align*}
    \alphahat &= \frac{4 \times 6.674\!\times\!10^{-11} \times
    1.989\!\times\!10^{30}}{(2.998\!\times\!10^{8})^2 \times
    6.96\!\times\!10^{8}} \\
    &= 8.49 \times 10^{-6}~\text{rad}
    = 8.49 \times 10^{-6} \times \frac{180 \times 3600}{\pi}
    = \boxed{1.75''}
\end{align*}
This is the value confirmed by Eddington's 1919 expedition.

\begin{annotation}[Scaling shortcut]
Since $\alphahat \propto M/b$, every point-mass deflection follows from the
solar one:
\[
\alphahat = 1.75''\,\frac{M}{\Msun}\,\frac{R_\odot}{b} .
\]
For $M = 10^{12}\,\Msun$ and $b = 10$~kpc,
$1.75''\times10^{12}\times(6.96\times10^8/3.086\times10^{20}) = 3.95''$,
the galaxy case below.
\end{annotation}


\textbf{(Galaxy lens.)} For $M = 10^{12}\,\Msun$ at $b = 10$~kpc
$= 3.086 \times 10^{20}$~m:
\[
    \alphahat = \frac{4 \times 6.674\!\times\!10^{-11} \times
    10^{12} \times 1.989\!\times\!10^{30}}
    {(2.998\!\times\!10^{8})^2 \times 3.086\!\times\!10^{20}}
    = 1.91 \times 10^{-5}~\text{rad}
    = \boxed{3.95''}
\]

\subsection*{Exercise~\ref{ex:superposition_prob}: Superposition of Deflections}

\textbf{(a)} Mass $M_1$ at $(-d/2, 0)$ and mass $M_2$ at $(+d/2, 0)$.
The photon passes at $\vecxi = (0, b)$.  The displacement vectors are:
\begin{align*}
    \vecxi - \vecxi_1 &= (d/2,\, b) \,, &
    |\vecxi - \vecxi_1|^2 &= d^2/4 + b^2 \,, \\
    \vecxi - \vecxi_2 &= (-d/2,\, b) \,, &
    |\vecxi - \vecxi_2|^2 &= d^2/4 + b^2 \,.
\end{align*}
The individual deflection angle vectors are:
\begin{align*}
    \boldsymbol{\hat{\alpha}}_1 &= \frac{4GM}{c^2}
    \frac{(d/2, b)}{d^2/4 + b^2} \,, \\
    \boldsymbol{\hat{\alpha}}_2 &= \frac{4GM}{c^2}
    \frac{(-d/2, b)}{d^2/4 + b^2} \,.
\end{align*}

\textbf{(b)} Adding the vectors:
\[
    \hat{\alpha}_x = \frac{4GM}{c^2(d^2/4 + b^2)}
    \left(\frac{d}{2} - \frac{d}{2}\right) = 0
\]
(the $x$-components cancel by symmetry).  The $y$-component:
\[
    \boxed{\hat{\alpha}_y = \frac{4GM}{c^2(d^2/4 + b^2)} \cdot 2b
    = \frac{8GMb}{c^2(d^2/4 + b^2)}}
\]

\textbf{(c)} In the limit $d \to 0$:
\[
    \hat{\alpha}_y \to \frac{8GMb}{c^2 b^2}
    = \frac{8GM}{c^2 b}
    = \frac{4G(2M)}{c^2 b}
\]
This is the deflection angle for a single mass of $2M$, as expected
from superposition when the two masses coincide.

\begin{derivation}[How fast the pair looks like a single mass (part b of the problem)]
Divide the pair result by the single-mass value $4G(2M)/(c^2b)$:
\[
\frac{\hat\alpha_y}{4G(2M)/(c^2b)} = \frac{8GMb}{c^2(d^2/4 + b^2)}\cdot\frac{c^2b}{8GM}
= \frac{b^2}{b^2 + d^2/4} = 1 - \frac{d^2}{4b^2} + \order{d^4/b^4} .
\]
The leading correction is $\order{d^2/b^2}$ (quadrupole).  There is no
$\order{d/b}$ (dipole) term because the expansion is about the centre of mass
of the pair.  At $b = 10\,d$ the ratio is $0.9975$.  This is the
statement behind treating a galaxy group, or a galaxy plus a satellite, as
a single deflector when seen from far enough away.
\end{derivation}
