\subsection*{Exercise~\ref{ex:soldner_full}: Soldner's Newtonian Deflection}

\textbf{(a)} For a particle with speed $v_\infty = \speedoflight$ and
impact parameter~$b$, the angular momentum is $L = b\speedoflight$ and the
energy per unit mass is $\mathcal{E} = \half \speedoflight^2$.  The
Newtonian orbit parameters are:
\begin{align*}
    p &= \frac{L^2}{\Gnewton M} = \frac{b^2 \speedoflight^2}{\Gnewton M} \,, \\
    e &= \sqrt{1 + \frac{2\mathcal{E} L^2}{(\Gnewton M)^2}}
      = \sqrt{1 + \frac{b^2 \speedoflight^4}{(\Gnewton M)^2}} \,.
\end{align*}


\begin{derivation}[Where $p$ and $e$ come from]
\textit{Orbit shape.}  With $u = 1/r$ and $L = r^2\dot\phi$, Newton's law
$\ddot{\mathbf r} = -\Gnewton M\hat{\mathbf r}/r^2$ becomes the Binet equation
$u'' + u = \Gnewton M/L^2$ ($' = d/d\phi$), solved by
$u = (\Gnewton M/L^2)(1 + e\cos\phi)$; hence $p = L^2/(\Gnewton M)$.

\textit{Eccentricity from energy.}  Using $\dot r = -L u'$,
$\mathcal{E} = \tfrac{L^2}{2}(u'^2 + u^2) - \Gnewton M u$.  Insert $u(\phi)$:
\[
    \mathcal{E} = \frac{(\Gnewton M)^2}{2L^2}\bigl[e^2\sin^2\phi
    + (1+e\cos\phi)^2\bigr] - \frac{(\Gnewton M)^2}{L^2}(1+e\cos\phi)
    = \frac{(\Gnewton M)^2}{2L^2}(e^2 - 1),
\]
since $e^2\sin^2\phi + e^2\cos^2\phi = e^2$ and the $2e\cos\phi$ terms
cancel.  Solving, $e^2 = 1 + 2\mathcal{E}L^2/(\Gnewton M)^2$.

\textit{Initial data.}  At infinity the corpuscle moves at $\speedoflight$ with
lever arm $b$, so $L = b\speedoflight$, and the potential vanishes, so
$\mathcal{E} = \tfrac12\speedoflight^2$.  Then
$2\mathcal{E}L^2 = b^2\speedoflight^4$.  (Checks A16a--b in
\texttt{problems\_02.wl}.)
\end{derivation}
\textbf{(b)} The orbit equation $r(\phi) = p/(1 + e\cos\phi)$ gives
$r \to \infty$ when $\cos\phi = -1/e$.  The two asymptotic angles are
$\phi_\pm = \pi \mp \delta$ where $\sin\delta = 1/e$.  The deflection
angle is:
\begin{equation*}
    \alphahat_{\mathrm{Newton}} = 2\delta = 2\arcsin(1/e) \,.
\end{equation*}
For $b \gg \Gnewton M / \speedoflight^2$, we have $e \gg 1$, so
$\delta \approx 1/e \approx \Gnewton M / (b \speedoflight^2)$ and:
\begin{equation*}
    \alphahat_{\mathrm{Newton}} \approx \frac{2\Gnewton M}{\speedoflight^2\, b} \,.
\end{equation*}


\begin{derivation}[The asymptotic angles, carefully]
$r \to \infty$ requires $1 + e\cos\phi = 0$.  The two solutions closest to
periapse ($\phi = 0$) are $\phi_\infty = \pm\arccos(-1/e)$.  Since
$\arccos(-x) = \pi/2 + \arcsin x$,
\[
    \phi_\infty = \pm\Bigl(\frac{\pi}{2} + \delta\Bigr), \qquad \sin\delta = \frac1e .
\]
The ray comes in along one asymptote and leaves along the other; the polar angle
swept is $2(\pi/2 + \delta) = \pi + 2\delta$.  An undeflected ray sweeps exactly
$\pi$, so the deflection is $2\delta = 2\arcsin(1/e)$.  (For $e \to \infty$,
$\delta \to 0$ and the orbit degenerates to the straight line $r\cos\phi = p/e
= b$.)  Check A16c verifies $\arccos(-1/e) = \pi/2 + \arcsin(1/e)$.
\end{derivation}
\textbf{(c)} For the Sun with $b = \Rsun$:
\begin{equation*}
    \alphahat_{\mathrm{Newton}} = \frac{2 \times 6.674 \times 10^{-11}
    \times 1.989 \times 10^{30}}{(2.998 \times 10^{8})^2 \times
    6.957 \times 10^{8}}
    \approx 4.24 \times 10^{-6}~\text{rad}
    = 0.875'' \,.
\end{equation*}


\begin{annotation}[Rounding]
To four figures the Newtonian value is $4.246\times10^{-6}$~rad
$= 0.876''$, exactly half of the GR value $1.75''$ computed with the same
constants (check A16d); the textbook ``$0.875''$'' is half of the rounded
$1.75''$.
\end{annotation}

\subsection*{Exercise~\ref{ex:factor_of_two}: The Factor of 2}

\textbf{(a)} From $\Delta\phi = \pi + 4m/r_0$ with
$m = \Gnewton M / \speedoflight^2$ and $r_0 \approx b$:
\begin{equation*}
    \alphahat = \Delta\phi - \pi = \frac{4m}{r_0}
    \approx \frac{4\Gnewton M}{\speedoflight^2\, b} \,.
\end{equation*}


\begin{annotation}[Why $r_0 \approx b$ is enough]
$r_0 = b - m - \tfrac32 m^2/b + \cdots$ (check A3), so
$4m/r_0 = 4m/b + 4m^2/b^2 + \cdots$.  The correction is second order, the same
order as the terms dropped in $\Delta\phi = \pi + 4m/r_0$; at first order the
replacement is exact.
\end{annotation}
\textbf{(b)} In the weak-field limit, the Schwarzschild metric in
isotropic coordinates takes the form:
\begin{equation*}
    ds^2 \approx -\!\left(1 - \frac{2\Gnewton M}{\speedoflight^2 r}\right)
    \speedoflight^2\, dt^2
    + \left(1 + \frac{2\Gnewton M}{\speedoflight^2 r}\right)
    (dx^2 + dy^2 + dz^2) \,.
\end{equation*}
For a null geodesic ($ds^2 = 0$), the effective refractive index is:
\begin{equation*}
    n(r) \approx \frac{1 + 2\Gnewton M / (\speedoflight^2 r)}
    {1 - 2\Gnewton M / (\speedoflight^2 r)}
    \approx 1 + \frac{4\Gnewton M}{\speedoflight^2 r} \,.
\end{equation*}
The Newtonian calculation effectively uses $n = 1 + 2\Gnewton M /
(\speedoflight^2 r)$ (only the $g_{tt}$ contribution), yielding half the
deflection.  The spatial curvature ($g_{rr}$) contributes the other factor
of~$2$.  Physically: in the Newtonian picture only time runs slower near a
mass, but in GR space is also stretched, so the photon path through curved
space is additionally bent.


\begin{derivation}[The refractive index, keeping the square root]
Null condition in the isotropic weak-field metric, with $m = \Gnewton
M/\speedoflight^2$:
\[
    (1 - 2m/r)\,\speedoflight^2dt^2 = (1 + 2m/r)\,|d\mathbf{x}|^2
    \;\Longrightarrow\;
    \Bigl|\frac{d\mathbf x}{dt}\Bigr| = \speedoflight\sqrt{\frac{1-2m/r}{1+2m/r}} .
\]
So $n = \speedoflight/|d\mathbf x/dt| = [(1+2m/r)/(1-2m/r)]^{1/2}
= 1 + 2m/r + \order{m^2/r^2}$ (check A15a), in agreement with Module~1e,
eq.~\eqref{eq:refractive_index}: $n = 1 - 2\Phi/\speedoflight^2$.  Each metric
component contributes $m/r$.

\textit{Deflection from $n$.}  A ray in a medium bends toward higher $n$ at the
rate $|\nabla_\perp n|$ per unit length.  Integrating along the unperturbed
straight line at impact parameter $b$, with $r = \sqrt{b^2+l^2}$:
\[
    \alphahat = \int_{-\infty}^{\infty}\Bigl|\frac{\partial}{\partial b}\,
    \frac{2m}{\sqrt{b^2+l^2}}\Bigr|\,dl = \frac{4m}{b}
    \qquad(\text{check A15b}).
\]
Keeping only the $g_{tt}$ part, $n - 1 = m/r$, the same integral gives
$2m/b$, Soldner's result (check A15c).  This is the precise sense in which
spatial curvature ``contributes the other half''.
\end{derivation}

\subsection*{Exercise~\ref{ex:deflections_numerical}: Numerical Deflections}

Using $\alphahat = 4\Gnewton M / (\speedoflight^2 b)$:

\textbf{(a) Sun:}
$\alphahat = 4 \times 6.674 \times 10^{-11} \times 1.989 \times 10^{30}
/ [(2.998 \times 10^{8})^2 \times 6.957 \times 10^{8}]
= 8.49 \times 10^{-6}~\text{rad} = 1.75''$.

\textbf{(b) Jupiter:}
$\alphahat = 4 \times 6.674 \times 10^{-11} \times 1.898 \times 10^{27}
/ [(2.998 \times 10^{8})^2 \times 7.149 \times 10^{7}]
= 7.88 \times 10^{-8}~\text{rad} = 0.016'' = 16~\text{mas}$.

\textbf{(c) Galaxy ($M = 10^{12}\,\Msun$, $b = 10$~kpc):}
$b = 10 \times 3.086 \times 10^{19}~\text{m} = 3.086 \times 10^{20}$~m.
$\alphahat = 4 \times 6.674 \times 10^{-11} \times 10^{12} \times
1.989 \times 10^{30} / [(2.998 \times 10^{8})^2 \times
3.086 \times 10^{20}] \approx 3.95''$.


\begin{annotation}[One-line check of all three]
$\alphahat = 2\RS/b$ with $\RS = 2.95$~km per solar mass.  Sun:
$2\times2.95~\text{km}/6.957\times10^5~\text{km} = 8.49\times10^{-6}$.
Galaxy: $\RS = 2.95\times10^{15}$~m, so
$2\times2.95\times10^{15}/3.086\times10^{20} = 1.91\times10^{-5}$~rad
$= 3.95''$.  (Checks A12a--d.)
\end{annotation}

\subsection*{Exercise~\ref{ex:shapiro_radar}: Shapiro Time Delay}

\textbf{(a)} The round-trip time is $\Delta T_{\mathrm{round}} =
2[T(R_\oplus) + T(R_{\mathrm{Merc}})] - 2(R_\oplus + R_{\mathrm{Merc}})/
\speedoflight$.  Using eq.~\eqref{eq:travel_time_expanded} and keeping only the leading logarithmic
term:
\begin{equation*}
    \Delta T \approx \frac{4\Gnewton \Msun}{\speedoflight^3}
    \left[\ln\!\left(\frac{4\, R_\oplus\, R_{\mathrm{Merc}}}{r_0^2}\right)
    + 1\right] .
\end{equation*}
The $+1$ comes from the non-logarithmic correction terms in the
weak-field expansion.


\begin{derivation}[The ``$+1$'', explicitly]
\textit{One leg.}  From eq.~\eqref{eq:travel_time_expanded}, the excess over the
flat-space straight-line time $\sqrt{R^2-r_0^2}/\speedoflight$ is
\[
    \delta T(R) = \frac{2m}{\speedoflight}\ln\frac{R + \sqrt{R^2-r_0^2}}{r_0}
    + \frac{m}{\speedoflight}\sqrt{\frac{R-r_0}{R+r_0}} .
\]
For $R \gg r_0$: $\sqrt{R^2-r_0^2} \to R$ inside the log and the square root
tends to~1, so $\delta T(R) \to (2m/\speedoflight)\ln(2R/r_0) + m/\speedoflight$
(check A14a).  The $m/\speedoflight$ is the non-logarithmic piece.

\textit{Earth $\to$ Mercury.}  The ray goes from $R_\oplus$ in to $r_0$ and out
to $R_{\rm Merc}$: $\delta T(R_\oplus) + \delta T(R_{\rm Merc})
= (2m/\speedoflight)[\ln(2R_\oplus/r_0) + \ln(2R_{\rm Merc}/r_0)]
+ 2m/\speedoflight$.

\textit{Round trip.}  Twice that.  Combining logs,
$\ln(2R_\oplus/r_0) + \ln(2R_{\rm Merc}/r_0) = \ln(4R_\oplus R_{\rm Merc}/r_0^2)$:
\[
    \Delta T = \frac{4m}{\speedoflight}\Bigl[\ln\frac{4R_\oplus R_{\rm Merc}}{r_0^2}
    + 1\Bigr], \qquad \frac{4m}{\speedoflight} = \frac{4\Gnewton\Msun}{\speedoflight^3}
    \qquad(\text{check A14b}).
\]
Each of the four one-way legs contributes $m/\speedoflight$, and
$4m/\speedoflight$ times~1 is the ``$+1$''.
\end{derivation}
\textbf{(b)} Numerically:
\begin{align*}
    \frac{4\Gnewton \Msun}{\speedoflight^3}
    &= \frac{4 \times 6.674 \times 10^{-11} \times 1.989 \times 10^{30}}
    {(2.998 \times 10^{8})^3}
    = 1.970 \times 10^{-5}~\text{s} = 19.7~\mu\text{s} \,.
\end{align*}
The logarithmic factor:
\begin{equation*}
    \ln\!\left(\frac{4 \times 1.496 \times 10^{11} \times
    5.79 \times 10^{10}}{(6.957 \times 10^{8})^2}\right)
    = \ln(7.15 \times 10^4) \approx 11.2 \,.
\end{equation*}
Therefore:
$\Delta T \approx 19.7 \times (11.2 + 1) = 240~\mu$s.


\begin{annotation}[How much the $+1$ matters]
$4\Gnewton\Msun/\speedoflight^3 = 19.7~\mu$s and the log is $11.18$; with the
$+1$ the delay is $240~\mu$s, without it $220~\mu$s (checks A14c--f).  The
non-logarithmic term is a 9\% effect, not negligible at the precision of the
radar experiments.
\end{annotation}
\textbf{(c)} Shapiro's 1964 measurement gave $200 \pm 20~\mu$s.  The
small discrepancy with our idealized point-mass calculation is consistent
given that: (i) Mercury is not exactly at inferior conjunction, (ii) the
Sun's corona introduces additional delays, and (iii) later measurements
with the Viking Mars lander confirmed the GR prediction to $\sim 0.1\%$
accuracy.

\subsection*{Exercise~\ref{ex:strong_field}: Exact vs.\ Weak-Field Deflection}

\textbf{(a)--(c)} Using numerical integration (in $G = c = M = 1$ units
where $\RS = 2$, so $m = 1$ and $b = 2 \times b/\RS$):

\begin{center}
\begin{tabular}{rcccc}
\toprule
$b/\RS$ & $\alphahat_{\mathrm{exact}}$~(rad) &
$\alphahat_1 = 4m/b$~(rad) &
$\alphahat_2$~(rad) &
Error($\alphahat_1$) \\
\midrule
5   & 0.5904  & 0.4000 & 0.5178 & $32\%$ \\
10  & 0.2361  & 0.2000 & 0.2295 & $15\%$ \\
20  & 0.1081  & 0.1000 & 0.1074 & $7.5\%$ \\
50  & 0.04122 & 0.04000 & 0.04118 & $3.0\%$ \\
100 & 0.02030 & 0.02000 & 0.02029 & $1.5\%$ \\
\bottomrule
\end{tabular}
\end{center}

The errors are substantial even at moderate $b/\RS$ because the expansion
parameter is $\varepsilon = m/r_0$, and $r_0 < b$ due to gravitational
attraction, so $\varepsilon > m/b$.  The second-order approximation
$\alphahat_2 = 4m/b + (15\pi/4)(m/b)^2$ is significantly more accurate.

\textbf{(d)} The first-order approximation achieves $1\%$ accuracy at
approximately $b/\RS \approx 150$, as confirmed by numerical integration.
The relatively slow convergence is because the expansion parameter
$\varepsilon = m/r_0$ is larger than the naive estimate $m/b = 1/(2\,b/\RS)$
due to the nonlinear relationship between $r_0$ and $b$.

\begin{derivation}[Second order, and a closed-form 1\% criterion]
\textit{In $\varepsilon = m/r_0$.}  The $\varepsilon^2$ term of the expanded
integrand is $\tfrac32\varepsilon^2(1-u^3)^2/(1-u^2)^{5/2}$; twice its integral
over $[0,1]$ is $15\pi/4 - 4$ (check A9a), so
$\alphahat = 4\varepsilon + (15\pi/4 - 4)\varepsilon^2 + \cdots$.

\textit{In $m/b$.}  Substituting $r_0 = b - m - \tfrac32m^2/b$:
$\alphahat = 4m/b + (15\pi/4)(m/b)^2 + \cdots$ (check A9c); the $-4$ is
cancelled by the $r_0 \to b$ conversion of the first-order term.  This is the
quantitative content of ``$\varepsilon > m/b$''.

\textit{1\% criterion.}  The relative size of the second-order term is
$(15\pi/16)(m/b)$, and with $m/b = 1/(2\,b/\RS)$ this equals
$(15\pi/32)/(b/\RS)$.  Setting it to $0.01$ gives $b/\RS \approx 147$,
consistent with the numerical crossing between $140$ ($1.05\%$) and $150$
($0.98\%$) (checks A10a--b).
\end{derivation}
