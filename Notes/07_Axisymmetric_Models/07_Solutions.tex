\subsection*{Exercise~\ref{ex:sis_deflection_derivation}: Derivation of the SIS Deflection Angle}

\textbf{(a)} Starting from $\rho(r) = \sigma_v^2 / (2\pi\Gnewton r^2)$
with $r = \sqrt{\xi^2 + z^2}$:
\begin{align*}
    \Sigma(\xi) &= \int_{-\infty}^{+\infty}
    \frac{\sigma_v^2}{2\pi\Gnewton\,(\xi^2 + z^2)}\, dz \,.
\end{align*}
Substituting $z = \xi\tan\phi$, $dz = \xi\sec^2\phi\,d\phi$,
$\xi^2 + z^2 = \xi^2\sec^2\phi$:
\begin{align*}
    \Sigma(\xi) &= \frac{\sigma_v^2}{2\pi\Gnewton}
    \int_{-\pi/2}^{\pi/2}
    \frac{\xi\sec^2\phi}{\xi^2\sec^2\phi}\, d\phi
    = \frac{\sigma_v^2}{2\pi\Gnewton\,\xi}
    \int_{-\pi/2}^{\pi/2} d\phi
    = \frac{\sigma_v^2}{2\Gnewton\,\xi} \,. \quad \checkmark
\end{align*}

\begin{derivation}[The substitution in (a), line by line]
With $z = \xi\tan\phi$: $dz = \xi\sec^2\phi\,d\phi$ and
$\xi^2+z^2 = \xi^2(1+\tan^2\phi) = \xi^2\sec^2\phi$.  The limits
$z\to\pm\infty$ map to $\phi\to\pm\pi/2$.  The integrand becomes
$\xi\sec^2\phi/(\xi^2\sec^2\phi) = 1/\xi$, and $\int_{-\pi/2}^{\pi/2}d\phi = \pi$.
The $\pi$ cancels the $\pi$ in $2\pi\Gnewton$, leaving $\sigma_v^2/(2\Gnewton\xi)$
(\texttt{problems\_07.wl} B4a).
\end{derivation}

\begin{annotation}[Dimensional shortcut]
The only quantities available are $\sigma_v$, $\Gnewton$ and $\xi$.  The
combination $\sigma_v^2/\Gnewton$ is a mass per length, so a surface density
must be $\propto\sigma_v^2/(\Gnewton\xi)$; only the number $1/2$ requires the
integral.  The same argument gives $M(\xi)\propto\sigma_v^2\xi/\Gnewton$ and a
dimensionless $\alphahat\propto\sigma_v^2/\speedoflight^2$.
\end{annotation}

\textbf{(b)} The enclosed projected mass:
\begin{equation*}
    M(\xi) = 2\pi \int_0^\xi \frac{\sigma_v^2}{2\Gnewton\,\xi'}\,
    \xi'\, d\xi'
    = \frac{\pi\sigma_v^2}{\Gnewton} \int_0^\xi d\xi'
    = \frac{\pi\sigma_v^2\,\xi}{\Gnewton} \,.
\end{equation*}

\textbf{(c)} The physical deflection angle:
\begin{equation*}
    \alphahat(\xi) = \frac{4\Gnewton M(\xi)}{\speedoflight^2\,\xi}
    = \frac{4\Gnewton}{\speedoflight^2\,\xi}\,
    \frac{\pi\sigma_v^2\,\xi}{\Gnewton}
    = \frac{4\pi\sigma_v^2}{\speedoflight^2} \,.
\end{equation*}
This is independent of $\xi$. \checkmark

\textbf{(d)} The reduced deflection angle is
$\alpha = (\Dds/\Ds)\,\alphahat = 4\pi(\sigma_v/\speedoflight)^2\,
\Dds/\Ds$.  Setting $\alpha = \thetaE$ (the constant deflection equals
the Einstein radius):
\begin{equation*}
    \thetaE = 4\pi\left(\frac{\sigma_v}{\speedoflight}\right)^2
    \frac{\Dds}{\Ds} \,. \quad \checkmark
\end{equation*}

\begin{annotation}[Why ``$\alpha = \thetaE$'' is the Einstein-radius condition]
The ring is the solution of $\beta = 0 = \theta-\alpha(\theta)$.  Because the
SIS $\alpha$ is constant, $\theta = \alpha$ at every radius, so the ring sits
at $\theta = \alpha$.  Equivalently $\bar\kappa(\thetaE) = 1$, the general
definition (main-text annotation on eq.~\eqref{eq:alpha_mean_kappa}).
\end{annotation}

\subsection*{Exercise~\ref{ex:sis_kappa_equals_gamma}: SIS --- $\kappab = |\gammaL|$ Everywhere}

\textbf{(a)} The mean convergence within $\theta$:
\begin{equation*}
    \bar{\kappa}(\theta) = \frac{2}{\theta^2}
    \int_0^\theta \frac{\thetaE}{2\theta'}\, \theta'\, d\theta'
    = \frac{2}{\theta^2}\, \frac{\thetaE}{2}\, \theta
    = \frac{\thetaE}{\theta} \,.
\end{equation*}

\textbf{(b)} The shear magnitude:
\begin{equation*}
    |\gammaL(\theta)| = \bar{\kappa}(\theta) - \kappab(\theta)
    = \frac{\thetaE}{\theta} - \frac{\thetaE}{2\theta}
    = \frac{\thetaE}{2\theta}
    = \kappab(\theta) \,. \quad \checkmark
\end{equation*}

\textbf{(c)} For the point mass, all the mass is concentrated at the
origin (a delta function), so $\kappab = 0$ everywhere except at
$\theta = 0$.  The shear $|\gammaL| = \thetaE^2/\theta^2 \neq 0$
arises from the tidal field of the central point mass.  For the SIS,
mass is distributed continuously (with $\Sigma \propto 1/\xi$), so
there is a nonzero convergence at every point.  The $1/r^2$ density
profile of the SIS is precisely the profile for which the local
convergence equals the shear at every radius.

\textbf{(d)} For $\kappab(\theta) = A\,\theta^{-n}$ with $A$ constant:
\begin{equation*}
    \bar{\kappa}(\theta) = \frac{2}{\theta^2}
    \int_0^\theta A\,\theta'^{-n}\, \theta'\, d\theta'
    = \frac{2A}{\theta^2}\, \frac{\theta^{2-n}}{2-n}
    = \frac{2A}{(2-n)\,\theta^n}
    = \frac{2}{2-n}\, \kappab(\theta) \,.
\end{equation*}
(Valid for $n < 2$; for $n \geq 2$ the integral diverges.)
The shear is:
\begin{equation*}
    |\gammaL| = \bar{\kappa} - \kappab
    = \left(\frac{2}{2-n} - 1\right) \kappab
    = \frac{n}{2-n}\, \kappab \,.
\end{equation*}
Setting $|\gammaL| = \kappab$: $n/(2-n) = 1$, so $n = 1$.
This is precisely the SIS exponent ($\kappab \propto \theta^{-1}$).
\checkmark

\begin{derivation}[The integral in (d)]
$\int_0^\theta\theta'^{1-n}d\theta' = \theta^{2-n}/(2-n)$ converges at $0$
only if $2-n>0$.  Then $\bar\kappa = (2/\theta^2)A\,\theta^{2-n}/(2-n)
= [2/(2-n)]\,\kappab$.  For $n\geq2$ the mass inside any radius is infinite;
this is the 2D statement that $\rho\propto r^{-3}$ or steeper is too
cuspy to have a finite projected mass.
\end{derivation}

\begin{bridge}[The power-law family in Part V]
With $n = \gamma'-1$ (3D slope $\gamma'$), the shear-to-convergence ratio is
$|\gammaL|/\kappab = (\gamma'-1)/(3-\gamma')$, and normalising
$\bar\kappa(\thetaE) = 1$ gives $\kappab(\thetaE) = (3-\gamma')/2$
(\texttt{problems\_07.wl} B7).  Chapter~\ref{ch:power_law_point_mass} uses
$\gamma'$ and a central point mass as its two knobs.  This exercise is the
convergence/shear half of that analysis.
\end{bridge}

\subsection*{Exercise~\ref{ex:sis_numerical}: SIS Numerical Example}

\textbf{(a)} With $\sigma_v = 250$~km\,s$^{-1} = 2.5 \times 10^5$~m\,s$^{-1}$:
\begin{equation*}
    \thetaE = 4\pi\left(\frac{2.5 \times 10^5}{3 \times 10^8}\right)^2
    \times \frac{1055}{1650}
    \approx 4\pi \times 6.944 \times 10^{-7} \times 0.639
    \approx 5.58 \times 10^{-6}~\text{rad}
    \approx 1.15'' \,.
\end{equation*}
(This uses the exercise's distances $\Dds = 1055$~Mpc, $\Ds = 1650$~Mpc; an
earlier version of this solution used $1200/1700$ and got $1.27''$.)

\textbf{(b)} For $\beta = 0.5''$:

(i) Since $\beta = 0.5'' < \thetaE \approx 1.15''$, the source is
\textit{inside} the caustic. Two images form.

(ii) Image positions:
\begin{equation*}
    \theta_+ = \beta + \thetaE = 0.5'' + 1.15'' = 1.65'' \,, \qquad
    \theta_- = \beta - \thetaE = 0.5'' - 1.15'' = -0.65'' \,.
\end{equation*}

(iii) Magnifications:
\begin{equation*}
    \mu_+ = 1 + \frac{\thetaE}{\beta}
    = 1 + \frac{1.15}{0.5} = 3.30 \,, \qquad
    \mu_- = 1 - \frac{\thetaE}{\beta}
    = 1 - \frac{1.15}{0.5} = -1.30 \,.
\end{equation*}

(iv) Total magnification:
$|\mu_+| + |\mu_-| = 3.30 + 1.30 = 4.60 = 2\thetaE/\beta$.

\begin{annotation}[Which distances?]
The exercise statement gives $\Dd\approx920$, $\Ds\approx1650$,
$\Dds\approx1055$\,Mpc (flat $\Lambda$CDM, $\Hubble = 70$, $\Omegam = 0.3$;
\texttt{problems\_07.wl} B6a).  With those, $\Dds/\Ds = 0.639$ and
$\thetaE = 1.15''$ (B6c), the chapter's Example value.  Note also $\Ds-\Dd = 730$\,Mpc $\neq\Dds$ (B6b).
\end{annotation}

\textbf{(c)} Equal absolute magnification requires
$|1 + \thetaE/\beta| = |1 - \thetaE/\beta|$.  For $\beta < \thetaE$
(doubly imaged), $\mu_+ = 1 + \thetaE/\beta > 0$ and
$|\mu_-| = \thetaE/\beta - 1$.  Setting these equal:
$1 + \thetaE/\beta = \thetaE/\beta - 1$, giving $1 = -1$, which is
impossible.  Equal magnification is never achieved for the SIS ---
the primary image is always brighter than the secondary. \checkmark

\begin{annotation}[Flux ratio as a function of $\beta$]
$|\mu_+/\mu_-| = (\thetaE+\beta)/(\thetaE-\beta)$, which equals 1 only at $\beta = 0$,
where the two images merge into the ring.  For a doubly imaged SIS source the
flux ratio and the image-position asymmetry
$(\theta_+-|\theta_-|)/(\theta_++|\theta_-|) = \beta/\thetaE$ carry the same
information (B15b).  This is why the SIS is a poor model when the observed flux ratio
and positions disagree (Module~\ref{ch:elliptical_models} adds shear and
ellipticity).
\end{annotation}

\textbf{(d)} The point mass Einstein radius for $M = 10^{12}\,\Msun$
at $z_d = 0.3$, $z_s = 1$ is $\thetaE \approx 2.4''$ (from Module~4).
The SIS Einstein radius depends on $\sigma_v$ rather than total mass.
For $\sigma_v = 250$~km\,s$^{-1}$, $\thetaE \approx 1.15''$, about half
as large: the SIS has only $\approx 2.3\times10^{11}\,\Msun$ inside its
Einstein radius, and $M(<\thetaE)\propto\thetaE^2$.  (An earlier version
quoted $1.1''$ for the point mass and called the SIS ring slightly larger;
the comparison is the other way round.)  The SIS is a more realistic model because it
accounts for the extended mass distribution, whereas the point mass
concentrates all mass at a single point.

\begin{annotation}[Check of (d)]
Evaluating $\thetaE = \sqrt{4\Gnewton M\Dds/(\speedoflight^2\Dd\Ds)}$ for $M = 10^{12}\,\Msun$
with the exercise's distances (920/1650/1055\,Mpc) gives $\approx2.4''$
(\texttt{problems\_07.wl} B6d), so the point mass has the
\emph{larger} Einstein radius here.  The physical comparison still holds: the SIS needs no total mass, because its
$\thetaE$ depends only on $\sigma_v$.  An SIS with $\thetaE = 1.15''$ has projected
mass $\pi\Sigmacr\Dd^2\thetaE^2 = (\speedoflight^2/4\Gnewton)(\Dd\Ds/\Dds)\,\thetaE^2\approx2.3\times10^{11}\,\Msun$
inside $\thetaE$ (B15d).  Because $M(<\thetaE)\propto\thetaE^2$, a point mass of
$10^{12}\,\Msun$ needs a ring about twice as large.
\end{annotation}

\subsection*{Exercise~\ref{ex:nis_convergence}: NIS Convergence and SIS Limit}

\textbf{(a)} Using $\kappab(\theta) = \Sigma(\theta)/\Sigmacr$ with
$\Sigmacr = \speedoflight^2 \Ds / (4\pi\Gnewton \Dd \Dds)$:
\begin{equation*}
    \kappab(\theta) = \frac{\sigma_v^2}{2\Gnewton\,\Dd\,
    \sqrt{\theta^2 + \theta_c^2}}\,
    \frac{4\pi\Gnewton\,\Dd\,\Dds}{\speedoflight^2\,\Ds}
    = \frac{2\pi\sigma_v^2\,\Dds}{\speedoflight^2\,\Ds}\,
    \frac{1}{\sqrt{\theta^2 + \theta_c^2}} \,.
\end{equation*}
Since $\thetaE = 4\pi(\sigma_v/\speedoflight)^2\,\Dds/\Ds$, we get
$2\pi\sigma_v^2 \Dds/(\speedoflight^2 \Ds) = \thetaE/2$, and:
\begin{equation*}
    \kappab(\theta) = \frac{\thetaE}{2\sqrt{\theta^2 + \theta_c^2}}
    \,. \quad \checkmark
\end{equation*}

\textbf{(b)} As $\theta_c \to 0$:
$\sqrt{\theta^2 + \theta_c^2} \to \sqrt{\theta^2} = \theta$ (for
$\theta > 0$), so $\kappab \to \thetaE/(2\theta)$, the SIS result.
\checkmark

\textbf{(c)} Multiple images require $\kappab(0) > 1$:
\begin{equation*}
    \kappab(0) = \frac{\thetaE}{2\theta_c} > 1
    \qquad \Longrightarrow \qquad
    \theta_c < \frac{\thetaE}{2} \,.
\end{equation*}
The inequality is strict (an earlier version wrote $\geq$): at
$\theta_c = \thetaE/2$ the radial critical curve has zero radius and every
source with $\beta\neq0$ has a single image.

\begin{derivation}[Why $\kappab(0)$ is the right quantity, and strict vs non-strict]
Multiple images on the axis require $\beta(\theta) = \theta-\alpha(\theta)$ to be
non-monotonic, i.e.\ $\lambda_r = 1-d\alpha/d\theta<0$ somewhere.  For the NIS,
$d\alpha/d\theta = \thetaE\theta_c/[s(s+\theta_c)]$, $s = \sqrt{\theta^2+\theta_c^2}$,
decreases monotonically from its central value $\thetaE/(2\theta_c) = \kappab(0)$
(\texttt{problems\_07.wl} B10d--f).  So multiple images
$\Leftrightarrow\kappab(0)>1\Leftrightarrow\theta_c<\thetaE/2$.  At exactly
$\theta_c = \thetaE/2$ the radial critical curve has zero radius and there is
one image for every $\beta\neq0$, so the critical value itself does
\emph{not} give multiple images, which is why the inequality is strict.
\end{derivation}

\textbf{(d)} For $\theta_c = 0.1\,\thetaE$:
\begin{align*}
    \kappab(0) &= \frac{\thetaE}{2 \times 0.1\,\thetaE} = 5.0 \,, \\
    \kappab(\thetaE) &= \frac{\thetaE}{2\sqrt{\thetaE^2 + 0.01\,\thetaE^2}}
    = \frac{1}{2\sqrt{1.01}} \approx 0.498 \,, \\
    \kappab(2\thetaE) &= \frac{1}{2\sqrt{4.01}} \approx 0.250 \,.
\end{align*}

\begin{annotation}[Reading the numbers]
To four places, $\kappab(\thetaE) = 0.4975$ and $\kappab(2\thetaE) = 0.2497$
(\texttt{problems\_07.wl} B13), within $0.5\%$ of the SIS values
$0.5$ and $0.25$ (B15c).  A core of $0.1\,\thetaE$ is invisible in $\kappab$ at the
Einstein radius.  It changes the centre completely, though: $\kappab(0) = 5$, a
radial critical curve at $0.251\,\thetaE$, and a radial caustic of radius
$0.427\,\thetaE$ (B10l).  Lensing at $\thetaE$ constrains the enclosed mass, while
the core shows up only through central and radial images.
\end{annotation}

\textbf{(e)} See the plot generated by
\texttt{problems\_07.wl}. As $\theta_c$ increases, the central
convergence decreases and the profile flattens in the core region,
while all profiles converge to the SIS behavior $\kappab = \thetaE/(2\theta)$
at large $\theta$.

\subsection*{Exercise~\ref{ex:nfw_concentration}: NFW Concentration and Lensing Strength}

\textbf{(a)} The function $f(x)$ has a logarithmic divergence as
$x \to 0$ (with $f(x) \simeq \ln(2/x) - 1$), peaks near $x \approx 0.6$
in terms of the product $x\,f(x)$, and falls off as $f(x) \sim 1/x^2$
for $x \gg 1$.  (An earlier version gave the small-$x$ form as
$(1/x)\ln(2/x)$, which is wrong; see the series in the main text.)  See the
plot generated by \texttt{problems\_07.wl}.

\begin{annotation}[Small-$x$ behaviour of $f$]
From $f = (1-K)/(x^2-1)$ and $K\simeq\ln(2/x)$ as $x\to0$,
$f(x)\simeq\ln(2/x)-1$: a pure logarithm, as stated in the main text
(\texttt{problems\_07.wl} B11i).  A $(1/x)\ln(2/x)$ divergence would be much
stronger and would make the projected mass inside small $x$ diverge.  The
large-$x$ statement
$f\sim1/x^2$ is right (B11k).
\end{annotation}

\textbf{(b)} For a cluster at $z_d = 0.3$ with $M_{200} = 10^{15}\,\Msun$
and $z_s = 1$: the virial radius is $r_{200} \approx 2.1$~Mpc, so
$r_s = r_{200}/c$, $\theta_s = r_s/\Dd$.
\begin{center}
\begin{tabular}{cccc}
\toprule
$c$ & $r_s$ (Mpc) & $\theta_s$ (arcsec) & $\kappa_s$ \\
\midrule
5 & 0.42 & 96 & 0.18 \\
10 & 0.21 & 48 & 0.50 \\
20 & 0.105 & 24 & 1.50 \\
\bottomrule
\end{tabular}
\end{center}
(Exact values depend on the cosmological parameters used; see
\texttt{problems\_07.wl}.)

\textbf{(c)} See the plot generated by \texttt{problems\_07.wl}.
Higher concentration pushes the $\kappab = 1$ crossing to larger
angular radii, increasing the Einstein radius and the strong-lensing
cross-section.

\textbf{(d)} (i)~The Einstein radius increases with $c$ because
$\kappa_s$ increases faster than $\theta_s$ decreases.
(ii)~The region where $\kappab > 1$ grows with $c$.
(iii)~The lensing cross-section for giant arcs (proportional to the
area enclosed by the tangential caustic) increases strongly with $c$,
roughly as $c^2$ for moderate concentrations.  This means that
concentration is a key parameter for predicting cluster arc
statistics. \checkmark

\begin{annotation}[Einstein radius is $\bar\kappa = 1$, not $\kappab = 1$]
Part (c) asks where $\kappab = 1$.  The Einstein radius is where the
\emph{mean} convergence reaches 1, $\bar\kappa(x) = (2\kappa_s/x^2)[\ln(x/2)+g(x)] = 1$
(eq.~\eqref{eq:nfw_alpha}), and it is always larger than the $\kappab = 1$
radius for a profile that decreases outward.  For an axisymmetric lens the
tangential caustic is a point, so ``area enclosed by the tangential caustic''
should be read as the cross-section of the elliptical generalisation
(Module~\ref{ch:elliptical_models}).
\end{annotation}
