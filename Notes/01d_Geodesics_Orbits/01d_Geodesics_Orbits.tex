\chapter{Geodesics and Orbits in Schwarzschild Spacetime}
\label{ch:geodesics_orbits}

\begin{quote}
\textit{Together the conserved quantities $E$ and $L$ provide a convenient way
to understand the orbits of particles in the Schwarzschild geometry.}
\hfill --- Sean Carroll, \textit{Spacetime and Geometry}
\end{quote}

\noindent
In Chapter~\ref{ch:schwarzschild} we derived the Schwarzschild metric.  We now ask how
particles and light move in this spacetime.  The answer comes from
the geodesic equation, which we reduce to a one-dimensional effective
potential problem using Killing vectors and conservation laws.  This
chapter derives the effective potential for massive and massless particles,
identifies its characteristic features (the ISCO and the photon sphere), and ends with
Mercury's perihelion precession, one of the three classical tests of GR.

\section{Killing Vectors and Conservation Laws}
\label{sec:killing_vectors}

The Schwarzschild metric is independent of both $t$ and $\phi$, and
therefore possesses two \emph{Killing vectors} (Carroll Sec.~5.4):
\begin{equation}
    K^\mu = (\partial_t)^\mu = (1, 0, 0, 0) \quad \text{(time translation)},
    \label{eq:killing_t}
\end{equation}
\begin{equation}
    R^\mu = (\partial_\phi)^\mu = (0, 0, 0, 1) \quad \text{(axial rotation)}.
    \label{eq:killing_phi}
\end{equation}
For any geodesic $x^\mu(\lambda)$, the quantity $K_\mu\, dx^\mu/d\lambda$
is conserved along the path (Carroll eq.~5.54).  Lowering the index with
the metric gives the two conserved quantities (Carroll eqs.~5.61--5.62,
with $c$ restored as in Congdon \& Keeton eq.~3.76a):
\begin{equation}
    \boxed{
    E = \left(1 - \frac{R_S}{r}\right) c^2\,\frac{dt}{d\lambda}
    }
    \label{eq:conserved_E}
\end{equation}
\begin{equation}
    \boxed{
    L = r^2 \frac{d\phi}{d\lambda}
    }
    \label{eq:conserved_L}
\end{equation}
For massive particles ($\lambda = \tau$), $E$ and $L$ are the conserved
energy (including rest energy, so $E = c^2$ for a particle at rest at
infinity) and angular momentum \emph{per unit mass}.\footnote{Carroll
sets $c = 1$ and writes $E = (1 - R_S/r)\,dt/d\lambda$.  Earlier versions
of this chapter used that $c = 1$ form in eqs.~\eqref{eq:conserved_E}
and~\eqref{eq:normalization} while keeping explicit factors of $c$ in
eqs.~\eqref{eq:effective_potential} and~\eqref{eq:photon_orbit}, so the
two did not match.  The factor $c^2$ follows Congdon \& Keeton
eq.~3.76a; with it, one $E$ serves every equation of this chapter.}  For photons, they are
the energy and angular momentum of the photon (up to normalization of the
affine parameter).

Spherical symmetry also allows us to restrict all orbits to the
equatorial plane $\theta = \pi/2$ without loss of generality (Carroll
eq.~5.56).

\begin{derivation}[Lowering the index: where $E$ and $L$ come from]
We write $f(r) \equiv 1 - R_S/r$ and use the coordinate time $t$, so
that $g_{tt} = -fc^2$, $g_{rr} = f^{-1}$, $g_{\theta\theta} = r^2$,
$g_{\phi\phi} = r^2\sin^2\theta$, and all off-diagonal components vanish.
\begin{align*}
K_\mu &= g_{\mu\nu}K^\nu = g_{\mu t} = (-fc^2,\,0,\,0,\,0)
  && \text{(diagonal metric, } K^\nu = \delta^\nu_t\text{)} \\
K_\mu \frac{dx^\mu}{d\lambda} &= -fc^2\,\frac{dt}{d\lambda} \equiv -E
  && (\text{sign: } E > 0 \text{ if future-directed}) \\
R_\mu &= g_{\mu\phi} = (0,\,0,\,0,\,r^2\sin^2\theta)
  && (R^\nu = \delta^\nu_\phi) \\
R_\mu \frac{dx^\mu}{d\lambda} &= r^2\sin^2\theta\,\frac{d\phi}{d\lambda}
  \;\xrightarrow{\ \theta = \pi/2\ }\; r^2\frac{d\phi}{d\lambda} \equiv L .
\end{align*}
These quantities are constant because, with $u^\mu = dx^\mu/d\lambda$,
\[
\frac{d}{d\lambda}\bigl(K_\mu u^\mu\bigr)
= u^\nu\nabla_\nu\bigl(K_\mu u^\mu\bigr)
= u^\mu u^\nu\,\nabla_{(\nu}K_{\mu)} + K_\mu\,u^\nu\nabla_\nu u^\mu = 0 .
\]
The first term vanishes by Killing's equation $\nabla_{(\mu}K_{\nu)} = 0$
(only the symmetric part survives contraction with the symmetric
$u^\mu u^\nu$); the second vanishes by the geodesic equation
$u^\nu\nabla_\nu u^\mu = 0$.  Setting $c = 1$ gives Carroll's form
$E = f\,dt/d\lambda$.
\end{derivation}

\begin{derivation}[The same two conservation laws straight from the Christoffel symbols]
The conservation laws also follow directly from the $t$ and $\phi$
components of the geodesic equation.  For the Schwarzschild metric
(equatorial plane; the two symbols below do not involve $c$, since
$g_{tt} = -fc^2$ enters $\Gamma^t_{\ tr} = \tfrac12 g^{tt}\partial_r g_{tt}$
only through the ratio $\partial_r g_{tt}/g_{tt}$) the Christoffel symbols
with an upper $t$ or $\phi$ that survive at $\theta = \pi/2$,
$\dot\theta = 0$ are
\[
\christoffel{t}{t}{r} = \frac{R_S}{2r(r - R_S)} = \frac{f'(r)}{2f(r)} ,
\qquad
\christoffel{\phi}{r}{\phi} = \frac{1}{r} .
\]
\begin{align*}
\ddot t + 2\,\christoffel{t}{t}{r}\,\dot t\,\dot r &= 0
  && (\Gamma^t_{\ tr} = \Gamma^t_{\ rt}\text{: factor 2}) \\
f\ddot t + f'\,\dot r\,\dot t &= 0
  && (\times f,\ \text{use } 2f\Gamma^t_{\ tr} = f') \\
\frac{d}{d\lambda}\bigl(f\,\dot t\bigr) &= 0
  && (\text{chain rule: } df/d\lambda = f'\dot r) \\[4pt]
\ddot\phi + \frac{2}{r}\,\dot r\,\dot\phi &= 0
  && (\phi \text{ component}) \\
r^2\ddot\phi + 2r\,\dot r\,\dot\phi = \frac{d}{d\lambda}\bigl(r^2\dot\phi\bigr) &= 0
  && (\times r^2) .
\end{align*}
So $f\dot t$ and $L = r^2\dot\phi$ are the first integrals of these
two components, and $E = fc^2\dot t$ is $c^2$ times the first.
\end{derivation}

\begin{annotation}[Units: one normalisation of $E$ throughout]
With $t$ the coordinate time ($g_{tt} = -fc^2$), eq.~\eqref{eq:conserved_E}
defines $E = f\,c^2\,dt/d\lambda$ (Congdon \& Keeton eq.~3.76a).  For
$\lambda = \tau$ it is an energy per unit mass, J\,kg$^{-1}$ =
m$^2$\,s$^{-2}$, equal to $c^2$ for a particle at rest at infinity.  The
same $E$ is used in every section:
eq.~\eqref{eq:normalization} has right-hand side $-\epsilon c^2$, with
$\mathcal{E} = \tfrac12(E^2/c^2 - \epsilon c^2)$
(derivation box below), and for photons it enters eq.~\eqref{eq:photon_orbit} and $b = cL/E$
(Section~\ref{sec:null_geodesics_preview}).
Setting $c = 1$ recovers Carroll's forms, $E = f\,dt/d\lambda$ and
right-hand side $-\epsilon$.  Dropping the $c^2$ from $E$ while keeping
explicit $c$ elsewhere is the usual source of stray factors of $c$.  As a dimension
check, with $L = r^2\,d\phi/d\tau$ in m$^2$\,s$^{-1}$, every term of
$V_{\mathrm{eff}}$ is a specific energy, m$^2$\,s$^{-2}$; e.g.\
$[GML^2/(c^2r^3)] = (\mathrm{m^3\,s^{-2}})(\mathrm{m^4\,s^{-2}})/
(\mathrm{m^2\,s^{-2}}\cdot\mathrm{m^3}) = \mathrm{m^2\,s^{-2}}$.
\end{annotation}

\section{The Effective Potential}
\label{sec:effective_potential}

The normalization condition for the four-velocity gives
(Carroll eq.~5.63, with $c$ restored):
\begin{equation}
    -\left(1 - \frac{R_S}{r}\right)c^2\left(\frac{dt}{d\lambda}\right)^2
    + \left(1 - \frac{R_S}{r}\right)^{-1}\left(\frac{dr}{d\lambda}\right)^2
    + r^2\left(\frac{d\phi}{d\lambda}\right)^2 = -\epsilon c^2 \,,
    \label{eq:normalization}
\end{equation}
where $\epsilon = 1$ for massive particles (timelike geodesics) and
$\epsilon = 0$ for photons (null geodesics).

Substituting the conserved quantities~\eqref{eq:conserved_E}
and~\eqref{eq:conserved_L}, we obtain a radial equation
(Carroll eq.~5.65):
\begin{equation}
    \frac{1}{2}\left(\frac{dr}{d\lambda}\right)^2 + V_{\mathrm{eff}}(r)
    = \mathcal{E} \,,
    \label{eq:radial_equation}
\end{equation}
where the \emph{effective potential} is (Carroll eq.~5.66, with
additive constant $+\tfrac{1}{2}\epsilon$ absorbed into $\mathcal{E}$;
Congdon \& Keeton eq.~3.80):
\begin{equation}
    \boxed{
    V_{\mathrm{eff}}(r) = -\epsilon\,\frac{GM}{r}
    + \frac{L^2}{2r^2}
    - \frac{GML^2}{c^2 r^3}
    }
    \label{eq:effective_potential}
\end{equation}
and $\mathcal{E} = \half(E^2/c^2 - \epsilon c^2)$ is the effective ``energy.''

\begin{derivation}[From the normalisation condition to the radial equation, with $c$ restored]
The metric is $ds^2 = -f c^2dt^2 + f^{-1}dr^2 + r^2d\Omega^2$ with
$f = 1 - R_S/r = 1 - 2GM/(c^2r)$, in the equatorial plane; dots denote
$d/d\lambda$, $E \equiv fc^2\,\dot t$, $L \equiv r^2\dot\phi$, and the normalisation is
$g_{\mu\nu}\dot x^\mu\dot x^\nu = -\epsilon c^2$.
\begin{align*}
-fc^2\,\dot t^2 + f^{-1}\dot r^2 + r^2\dot\phi^2 &= -\epsilon c^2
  && \text{(normalisation)} \\
-\frac{E^2}{fc^2} + \frac{\dot r^2}{f} + \frac{L^2}{r^2} &= -\epsilon c^2
  && \bigl(\dot t = E/(fc^2),\ \dot\phi = L/r^2\bigr) \\
\dot r^2 &= \frac{E^2}{c^2} - f\Bigl(\epsilon c^2 + \frac{L^2}{r^2}\Bigr)
  && (\times f,\ \text{rearrange}) \\
&= \frac{E^2}{c^2} - \epsilon c^2 + \frac{\epsilon c^2R_S}{r} - \frac{L^2}{r^2}
   + \frac{R_SL^2}{r^3}
  && (\text{expand } f = 1 - R_S/r) \\
\half\dot r^2 &= \half\Bigl(\frac{E^2}{c^2} - \epsilon c^2\Bigr) + \frac{\epsilon GM}{r}
   - \frac{L^2}{2r^2} + \frac{GML^2}{c^2r^3}
  && (\div 2,\ R_S = 2GM/c^2) .
\end{align*}
Moving the last three terms to the left-hand side gives
eq.~\eqref{eq:radial_equation} with $V_{\mathrm{eff}}$ of
eq.~\eqref{eq:effective_potential} and
$\mathcal{E} = \tfrac12(E^2/c^2 - \epsilon c^2)$, the form of Congdon \&
Keeton eq.~3.78.
\end{derivation}

\begin{derivation}[Consistency check: the $r$-component of the geodesic equation]
Differentiating eq.~\eqref{eq:radial_equation} with respect to $\lambda$
gives $\dot r\,\ddot r + V_{\mathrm{eff}}'(r)\,\dot r = 0$, i.e.\
$\ddot r = -V_{\mathrm{eff}}'(r)$ wherever $\dot r \neq 0$.  The same
equation follows directly from the $r$-component of the geodesic equation
($c = 1$, $\theta = \pi/2$),
\[
\ddot r = -\christoffel{r}{t}{t}\,\dot t^2 - \christoffel{r}{r}{r}\,\dot r^2
          - \christoffel{r}{\phi}{\phi}\,\dot\phi^2 ,
\qquad
\christoffel{r}{t}{t} = \frac{R_S(r - R_S)}{2r^3},\quad
\christoffel{r}{r}{r} = -\frac{R_S}{2r(r - R_S)},\quad
\christoffel{r}{\phi}{\phi} = -(r - R_S) ,
\]
after substituting ($c = 1$, so $E = f\dot t$) $\dot t = E/f$, $\dot\phi = L/r^2$ and
$\dot r^2 = E^2 - f(\epsilon + L^2/r^2)$: the right-hand side reduces
identically to $-V_{\mathrm{eff}}'(r)$.  The first integral
eq.~\eqref{eq:radial_equation} therefore carries all of the radial dynamics.
\end{derivation}

\begin{annotation}[Reading the effective potential]
To see what $\mathcal{E}$ is, write $E = c^2 + e$, with $e$
the Newtonian specific energy (kinetic plus potential).  Then
$\mathcal{E} = \tfrac12(E^2/c^2 - c^2) = e + e^2/(2c^2)$, so
$\mathcal{E} \to e$ in the Newtonian limit.  This is why
absorbing Carroll's constant $\tfrac12\epsilon$ into $\mathcal{E}$
is convenient: $V_{\mathrm{eff}}$ then has the Newtonian terms
with their Newtonian normalisation.
The size of the GR term follows from the ratio of the third to the
second term,
$\bigl(GML^2/c^2r^3\bigr)/\bigl(L^2/2r^2\bigr) = 2GM/(c^2r) = R_S/r$.
``Newtonian limit'' therefore means $r \gg R_S$, not $c \to \infty$
as an abstract instruction.  At Mercury's orbit $R_S/a \approx
5.1\times10^{-8}$; at the ISCO $R_S/r = 1/3$.
As for its sign, the GR term enters with a minus sign for both
$\epsilon = 1$ and $\epsilon = 0$: it is attractive and grows
fastest as $r \to 0$.
\end{annotation}


\begin{remark}
The three terms in $V_{\mathrm{eff}}$ each have a physical interpretation:
\begin{enumerate}[nosep]
    \item $-\epsilon\, GM/r$ is the Newtonian gravitational potential
          (absent for photons since $\epsilon = 0$);
    \item $L^2/(2r^2)$ is the centrifugal barrier (identical to Newtonian
          mechanics);
    \item $-GML^2/(c^2 r^3)$ is the GR correction, a term which
          has no Newtonian analogue and is responsible for all the
          distinctive features of relativistic orbits.
\end{enumerate}
The GR term is attractive at small $r$, so the centrifugal
barrier can be overcome; this is why particles can spiral into black
holes, which is impossible in Newtonian gravity.
\end{remark}

\begin{figure}[ht]
    \centering
    \includegraphics[width=0.75\textwidth]{../Figures/01d_Geodesics_Orbits/newtonian_vs_gr.pdf}
    \caption{Comparison of Newtonian (dashed) and GR (solid) effective
    potentials for a massive particle with $L = 4\,GM/c$.  The GR correction
    term $\propto 1/r^3$ pulls the potential down at small $r$, allowing
    particles to plunge past the centrifugal barrier.
    (cf.\ Carroll Figs.~5.4 and 5.5)}
    \label{fig:newtonian_vs_gr}
\end{figure}

\section{Circular Orbits and the ISCO}
\label{sec:circular_orbits}

Circular orbits occur at extrema of the effective potential:
$dV_{\mathrm{eff}}/dr = 0$.  For massive particles ($\epsilon = 1$),
this gives (Carroll eq.~5.68):
\begin{equation}
    GM r_c^2 - L^2 r_c + 3GML^2/c^2 = 0 \,.
    \label{eq:circular_condition}
\end{equation}
The two solutions are (Carroll eq.~5.71):
\begin{equation}
    r_\pm = \frac{L^2}{2GM}
    \left(1 \pm \sqrt{1 - \frac{12G^2M^2}{L^2 c^2}}\right) \,.
    \label{eq:circular_radii}
\end{equation}

The outer solution $r_+$ is a \emph{stable} circular orbit (a minimum of
$V_{\mathrm{eff}}$) and the inner solution $r_-$ an \emph{unstable} one
(a maximum of $V_{\mathrm{eff}}$).
Real solutions require $L^2 \geq 12\,G^2M^2/c^2$, i.e.,
$L \geq 2\sqrt{3}\,GM/c$ (Carroll eq.~5.73).  The minimum stable circular
orbit occurs when $r_+ = r_-$, giving the \emph{innermost stable
circular orbit} (ISCO) (Carroll eq.~5.74; Congdon \& Keeton eq.~3.84):
\begin{equation}
    \boxed{r_{\mathrm{ISCO}} = \frac{6GM}{c^2} = 3\, R_S}
    \label{eq:ISCO}
\end{equation}

\begin{derivation}[Circular orbits, which one is stable, and the ISCO]
For $\epsilon = 1$:
\begin{align*}
\frac{dV_{\mathrm{eff}}}{dr}
  &= \frac{GM}{r^2} - \frac{L^2}{r^3} + \frac{3GML^2}{c^2r^4}
   = \frac{GMr^2 - L^2r + 3GML^2/c^2}{r^4}
  && \text{(differentiate term by term)} \\
0 &= GMr_c^2 - L^2r_c + \frac{3GML^2}{c^2}
  && (\times r_c^4) \\
r_\pm &= \frac{L^2 \pm \sqrt{L^4 - 12G^2M^2L^2/c^2}}{2GM}
  && \text{(quadratic formula)} \\
  &= \frac{L^2}{2GM}\left(1 \pm \sqrt{1 - \frac{12G^2M^2}{L^2c^2}}\right)
  && (\text{pull } L^2 \text{ out of the root}) .
\end{align*}
To decide which root is the minimum, note that the numerator factorises,
$GMr^2 - L^2r + 3GML^2/c^2 = GM(r - r_+)(r - r_-)$, so
$V_{\mathrm{eff}}' = GM(r - r_+)(r - r_-)/r^4$ and, differentiating the
product (every term still containing a factor $r - r_\pm$ vanishes at
$r = r_\pm$),
\[
V_{\mathrm{eff}}''(r_\pm) = \pm\,\frac{GM\,(r_+ - r_-)}{r_\pm^4} .
\]
This is positive at $r_+$ (minimum, stable) and negative at $r_-$
(maximum, unstable).  The two merge when the square root vanishes,
$L^2 = 12G^2M^2/c^2$, and then
$r = L^2/(2GM) = 12G^2M^2/(2GMc^2) = 6GM/c^2$.
\end{derivation}

\begin{annotation}[What the ISCO is, and the classic slip]
$r_+(L)$ increases monotonically with $L$ for
$L > 2\sqrt3\,GM/c$, so $6GM/c^2$ is the \emph{smallest} radius of
any stable circular orbit.  Meanwhile $r_-(L)$ decreases from
$6GM/c^2$ towards $3GM/c^2$ as $L \to \infty$: unstable circular
orbits of massive particles fill $3GM/c^2 < r < 6GM/c^2$ and
approach the photon sphere (Section~\ref{sec:photon_sphere}) in the
ultra-relativistic limit.  In the $G = c = M = 1$ units of the
Solutions, $L = 4$ gives $r_+ = 12$, $r_- = 4$.
A common slip is to write $r_{\mathrm{ISCO}} = 6GM/c^2 = 3R_S$ as
$6R_S$; mixing $GM/c^2$ and $R_S = 2GM/c^2$ is the most common
factor-of-2 error in this chapter.
The efficiency quoted in the remark below follows from the value at the ISCO
($G = c = M = 1$, $L^2 = 12$): $V_{\mathrm{eff}}(6) = -1/18$, so
$(E/c^2)^2 = 1 + 2V_{\mathrm{eff}}/c^2 = 8/9$ and $E/c^2 = 2\sqrt2/3$.  Gas
that drifts inward through circular orbits down to the ISCO has
radiated the fraction $1 - 2\sqrt2/3 \approx 0.057$ (5.7\%) of its
rest-mass energy.
\end{annotation}


\begin{remark}
The ISCO is a fundamental scale in black hole astrophysics.  Accreting
matter in a thin disk spirals inward until it reaches the ISCO, then
plunges into the black hole.  The ISCO radius determines the
efficiency of energy extraction from accretion and the inner edge of
accretion disks observed in X-ray binaries and active galactic nuclei.
\end{remark}

\section{The Photon Sphere}
\label{sec:photon_sphere}

For photons ($\epsilon = 0$), the effective potential simplifies to
(Congdon \& Keeton eq.~3.86):
\begin{equation}
    V_{\mathrm{eff}}^{\mathrm{photon}}(r) = \frac{L^2}{2r^2}
    - \frac{GML^2}{c^2 r^3} \,.
    \label{eq:Veff_photon}
\end{equation}
Setting $dV_{\mathrm{eff}}/dr = 0$ gives a single circular orbit
(Carroll eq.~5.70; Congdon \& Keeton Sec.~3.3.3, prose after eq.~3.86):
\begin{equation}
    \boxed{r_{\mathrm{photon}} = \frac{3GM}{c^2} = \frac{3}{2}\, R_S}
    \label{eq:photon_sphere}
\end{equation}
This is an \emph{unstable} circular orbit: any perturbation causes
the photon either to escape to infinity or to spiral into the black hole.
Nevertheless, the photon sphere governs strong gravitational
lensing near black holes and is directly related to the ``photon ring''
observed by the Event Horizon Telescope.

\begin{derivation}[Photon sphere with units, and the capture impact parameter]
\[
\frac{dV_{\mathrm{eff}}^{\mathrm{photon}}}{dr}
= -\frac{L^2}{r^3} + \frac{3GML^2}{c^2r^4}
= \frac{L^2}{r^4}\left(\frac{3GM}{c^2} - r\right) ,
\]
which vanishes only at $r = 3GM/c^2$, for \emph{every} $L$: unlike the
massive case, the circular photon orbit does not depend on angular
momentum.  The height of the barrier ($G = c = M = 1$) is
$V^{\mathrm{photon}}_{\mathrm{eff}}(3) = L^2/18 - L^2/27 = L^2/54$.
A photon from infinity has $\mathcal{E} = \tfrac12E^2/c^2 = \tfrac12E^2$
($\epsilon = 0$), so it turns around outside the barrier iff
$\tfrac12E^2 < L^2/54$, i.e.\ iff $b^2 = c^2L^2/E^2 = L^2/E^2 > 27$.  Restoring units,
\[
b_{\mathrm{crit}} = 3\sqrt3\,\frac{GM}{c^2} = \frac{3\sqrt3}{2}\,R_S
\approx 2.598\,R_S ;
\]
photons with $b < b_{\mathrm{crit}}$ are captured.  This is the capture
line in Fig.~\ref{fig:deflection_comparison} of
Chapter~\ref{ch:linearized_gravity}.
\end{derivation}

\begin{annotation}[How far is lensing from the photon sphere?]
``Strong'' lensing by galaxies is strong only in the sense of producing
multiple images; the null geodesics involved stay at $b \gg R_S$.  For the
Sun at the limb, $b/R_S = R_\odot/R_S \approx 2.36\times10^5$.  The photon
sphere and $b_{\mathrm{crit}}$ matter for black-hole imaging (the EHT
photon ring), never for galaxy-scale image positions.
\end{annotation}

\begin{bridge}[Chapter~\ref{ch:power_law_point_mass}: the central black hole is a weak-field point mass]
Part~V adds a central point mass (a supermassive black hole) to a
power-law galaxy.  For $M_\bullet = 10^9\,M_\odot$,
$R_S \approx 2.95\times10^{12}$~m $\approx 9.6\times10^{-5}$~pc, so a ray
at $b = 1$~kpc has $b/R_S \approx 1.0\times10^7$ (and still
$\approx 1.0\times10^6$ at $b = 100$~pc).  The photon sphere, the ISCO and
$b_{\mathrm{crit}}$ derived here lie six to seven orders of magnitude
inside the region probed by lensed images: the black hole enters the
compound-lens model only through its weak-field deflection $4GM_\bullet/(c^2b)$
(Chapter~\ref{ch:linearized_gravity}).
\end{bridge}

\section{Effective Potential: Newtonian vs.\ GR}
\label{sec:veff_comparison}

The qualitative differences between the Newtonian and GR effective
potentials are collected in the following table.

\begin{center}
\begin{tabular}{lcc}
\toprule
\textbf{Feature} & \textbf{Newtonian} & \textbf{GR} \\
\midrule
Centrifugal barrier & Diverges as $r \to 0$ & Overcome at small $r$ \\
ISCO & None (stable at any $r$) & $r = 3R_S$ \\
Photon orbits & None & Unstable at $r = 3R_S/2$ \\
Plunge orbits & Impossible & Possible for $r < r_{\mathrm{ISCO}}$ \\
\bottomrule
\end{tabular}
\end{center}

\section{Perihelion Precession}
\label{sec:perihelion_precession}

In Newtonian gravity, bound orbits are perfect ellipses that close on
themselves.  In GR, the extra $-GML^2/(c^2 r^3)$ term in the effective
potential causes the orbit to \emph{precess}: the perihelion (point of
closest approach) advances by a small angle each orbit.

Following Carroll Sec.~5.5 and using the substitution $x = L^2/(GMr)$,
the orbit equation becomes (Carroll eq.~5.79):
\begin{equation}
    \frac{d^2 x}{d\phi^2} + x = 1 + \frac{3G^2M^2}{L^2 c^2}\, x^2 \,.
    \label{eq:orbit_equation}
\end{equation}
The last term is the GR correction.  Treating it as a perturbation
($x = x_0 + x_1$, with $x_0 = 1 + e\cos\phi$ the Keplerian solution),
one finds the accumulated precession per orbit
(Carroll eq.~5.92):
\begin{equation}
    \boxed{
    \Delta\phi = \frac{6\pi G M}{c^2 a(1 - e^2)}
    = \frac{6\pi G^2 M^2}{c^2 L^2}
    }
    \label{eq:perihelion_precession}
\end{equation}
where $a$ is the semi-major axis and $e$ the eccentricity.

\begin{derivation}[From the radial equation to the orbit equation~\eqref{eq:orbit_equation}]
We use $u = 1/r$ and the angle $\phi$ as the parameter along the orbit.  With
$d\phi/d\lambda = L/r^2 = Lu^2$ and primes denoting $d/d\phi$,
\[
\frac{dr}{d\lambda} = \frac{dr}{d\phi}\,\frac{d\phi}{d\lambda}
= \left(-\frac{1}{u^2}\,u'\right)Lu^2 = -L\,u' .
\]
Eq.~\eqref{eq:radial_equation} with $\epsilon = 1$ becomes
\[
\tfrac12L^2u'^2 - GMu + \tfrac12L^2u^2 - \frac{GML^2}{c^2}\,u^3 = \mathcal{E} .
\]
Differentiating with respect to $\phi$ ($\mathcal{E}$ is constant) gives
\[
L^2u'\left(u'' + u - \frac{GM}{L^2} - \frac{3GM}{c^2}\,u^2\right) = 0 .
\]
Away from turning points ($u' \neq 0$),
$u'' + u = GM/L^2 + 3GMu^2/c^2$.  Multiplying by $L^2/(GM)$ and setting
$x = L^2u/(GM)$, i.e.\ $u = GMx/L^2$, we obtain
\[
x'' + x = 1 + \frac{3L^2}{c^2}\,u^2 = 1 + \frac{3G^2M^2}{L^2c^2}\,x^2 ,
\]
which is eq.~\eqref{eq:orbit_equation}.  Without the GR term,
$x_0 = 1 + e\cos\phi$, i.e.\ $r = (L^2/GM)/(1 + e\cos\phi)$: a conic with
semi-latus rectum $p = L^2/(GM)$.  Then
$r_{\mathrm{peri}} + r_{\mathrm{apo}} = p/(1 + e) + p/(1 - e) = 2a$ gives
$L^2 = GMa(1 - e^2)$.
\end{derivation}

\begin{derivation}[First-order perturbation theory and the precession angle]
We write $\varepsilon_{\mathrm{GR}} \equiv 3G^2M^2/(L^2c^2) \ll 1$ (not to be
confused with the timelike/null flag $\epsilon$) and
$x = x_0 + x_1$, $x_0 = 1 + e\cos\phi$, $x_1 = \order{\varepsilon_{\mathrm{GR}}}$.
To first order,
\[
x_1'' + x_1 = \varepsilon_{\mathrm{GR}}\,x_0^2
= \varepsilon_{\mathrm{GR}}\Bigl[\Bigl(1 + \tfrac{e^2}{2}\Bigr)
  + 2e\cos\phi + \tfrac{e^2}{2}\cos2\phi\Bigr]
\qquad \bigl(\cos^2\phi = \tfrac12(1 + \cos2\phi)\bigr).
\]
We solve term by term with the operator $d^2/d\phi^2 + 1$.
A constant $C$ is reproduced by $x_1 = C$.
Since $(d^2/d\phi^2 + 1)\cos2\phi = -3\cos2\phi$,
the term $\tfrac{e^2}{2}\cos2\phi$ needs $-\tfrac{e^2}{6}\cos2\phi$.
The term $2e\cos\phi$ is \emph{resonant} ($\cos\phi$ solves the homogeneous
equation); since $(d^2/d\phi^2 + 1)(\phi\sin\phi) = 2\cos\phi$, it
needs $e\,\phi\sin\phi$.  Altogether,
\[
x_1 = \varepsilon_{\mathrm{GR}}\Bigl[1 + \tfrac{e^2}{2}
      - \tfrac{e^2}{6}\cos2\phi + e\,\phi\sin\phi\Bigr] .
\]
Only the secular term $e\,\phi\sin\phi$ grows.  Since
$1 + e\cos[(1 - \varepsilon_{\mathrm{GR}})\phi]
= 1 + e\cos\phi + \varepsilon_{\mathrm{GR}}\,e\,\phi\sin\phi + \order{\varepsilon_{\mathrm{GR}}^2}$,
the orbit is $x \simeq 1 + e\cos[(1 - \varepsilon_{\mathrm{GR}})\phi]$
plus non-growing terms.  Perihelion (maximum of $x$) recurs when
$(1 - \varepsilon_{\mathrm{GR}})\phi = 2\pi n$, so per orbit $\phi$ advances by
$2\pi/(1 - \varepsilon_{\mathrm{GR}}) = 2\pi + 2\pi\varepsilon_{\mathrm{GR}}
+ \order{\varepsilon_{\mathrm{GR}}^2}$:
\[
\Delta\phi = 2\pi\varepsilon_{\mathrm{GR}} = \frac{6\pi G^2M^2}{L^2c^2}
= \frac{6\pi GM}{c^2a(1 - e^2)}
\qquad \bigl(L^2 = GMa(1 - e^2)\bigr).
\]
\end{derivation}


For Mercury (Carroll eq.~5.96) this gives
\begin{equation}
    \Delta\phi_{\mathrm{Mercury}} = 5.02 \times 10^{-7}~\text{rad/orbit}
    = 0.104''\text{/orbit} = 43.0''\text{/century} \,.
    \label{eq:mercury_precession}
\end{equation}
This matched the previously unexplained 43~arcseconds/century discrepancy
in Mercury's orbit and was one of Einstein's first confirmations of GR.

\begin{annotation}[Numbers and units for Mercury]
Everything reduces to one dimensionless number, $GM/(c^2a)$.  With
$GM_\odot/c^2 \approx 1477$~m and $a = 5.79\times10^{10}$~m,
$GM_\odot/(c^2a) \approx 2.55\times10^{-8}$; with $1 - e^2 = 0.9577$,
$\Delta\phi = 6\pi\times2.55\times10^{-8}/0.9577 = 5.02\times10^{-7}$~rad per
orbit $= 0.1036''$ per orbit.  A Julian century holds
$36525/87.97 = 415.2$ orbits, giving $43.0''$ per century.  Carry four
significant figures through the last multiplication:
$0.1036\times415.2 = 43.0$, whereas a per-orbit value of $0.1034''$ (a slip,
not a rounding of $0.10355''$) would give $42.9''$.
\end{annotation}

\section{Null Geodesics and the Deflection of Light (Preview)}
\label{sec:null_geodesics_preview}

For photons, the orbit equation (using $u = 1/r$) becomes
(Congdon \& Keeton eq.~3.87):
\begin{equation}
    \frac{d\phi}{dr} = \pm \frac{L}{r^2}
    \left[\frac{E^2}{c^2} - \frac{L^2}{r^2}\left(1 - \frac{R_S}{r}\right)
    \right]^{-1/2} \,.
    \label{eq:photon_orbit}
\end{equation}
A photon approaching a mass $M$ with impact parameter $b = cL/E$
(the perpendicular distance from the mass in the absence of deflection)
will be deflected by an angle (Congdon \& Keeton eq.~3.96):
\begin{equation}
    \hat{\alpha} = \frac{4GM}{c^2 b} \,,
    \label{eq:deflection_preview}
\end{equation}
in the weak-field limit ($b \gg R_S$).  This is \emph{twice} the
Newtonian prediction and was confirmed by Eddington's 1919 solar eclipse
observations.

The full derivation of the deflection angle is the subject of Chapter~\ref{ch:light_deflection}.
We mention it here because the effective potential framework shows
why light bends: photons follow null geodesics of the
curved Schwarzschild spacetime, and the GR correction term in the
effective potential deflects them more strongly than Newtonian gravity
would predict.

\begin{derivation}[From eq.~\eqref{eq:photon_orbit} to $\hat\alpha = 4GM/(c^2b)$]
With $E = fc^2\,dt/d\lambda$ (eq.~\eqref{eq:conserved_E}) and
$\epsilon = 0$, the radial equation reads
$\dot r^2 = E^2/c^2 - fL^2/r^2$, so $dr/d\lambda \to E/c$ at infinity and $b = L/|dr/d\lambda|_\infty = cL/E$.
We set $u = 1/r$, $dr = -du/u^2$, and square eq.~\eqref{eq:photon_orbit}:
\[
\left(\frac{du}{d\phi}\right)^2 = u^4\left(\frac{dr}{d\phi}\right)^2
= \frac{E^2}{c^2L^2} - u^2(1 - R_Su)
= \frac{1}{b^2} - u^2 + R_Su^3 .
\]
Differentiating with respect to $\phi$ and dividing by $2u'$ gives
\[
u'' + u = \tfrac32R_S\,u^2 = \frac{3GM}{c^2}\,u^2 .
\]
At zeroth order ($R_S \to 0$) the solution is the straight line $u_0 = \sin\phi/b$
(i.e.\ $r\sin\phi = b$).  At first order the source is
$\tfrac32R_Su_0^2 = \tfrac{3R_S}{4b^2}(1 - \cos2\phi)$; using
$(d^2/d\phi^2 + 1)\cos2\phi = -3\cos2\phi$ the particular solution is
\[
u_1 = \frac{3R_S}{4b^2} + \frac{R_S}{4b^2}\cos2\phi
    = \frac{R_S}{2b^2}\bigl(1 + \cos^2\phi\bigr) .
\]
The asymptotes are where $u = u_0 + u_1 = 0$, at $\phi = -\delta$ and at
$\phi = \pi + \delta$; near $\phi = 0$,
$-\delta/b + R_S/b^2 = 0$, so $\delta = R_S/b$ at each end.  The total
swing beyond the straight-line value $\pi$ is $2\delta$:
\[
\hat\alpha = \frac{2R_S}{b} = \frac{4GM}{c^2b} .
\]
Chapter~\ref{ch:light_deflection} obtains the same result from the
integral form of the orbit equation.
\end{derivation}

\begin{annotation}[Where the ``factor of two'' lives in this picture]
For photons $\epsilon = 0$ removes the $-GM/r$ term altogether: in
Schwarzschild coordinates there is no ``Newtonian part'' of the photon
potential, and the whole $4GM/(c^2b)$ comes from the single $1/r^3$ term
($\tfrac32R_Su^2$ in the orbit equation).  The split ``half from $g_{tt}$,
half from $g_{ij}$'' of Chapter~\ref{ch:linearized_gravity} is made in
isotropic coordinates.  The split depends on the coordinates; the total
deflection does not.  Do not identify the $1/r^3$ term with the
``spatial-curvature half''.  The Newtonian corpuscle value, $2GM/(c^2b)$, is
derived in Section~\ref{sec:soldner}.
\end{annotation}

\begin{bridge}[Chapter~\ref{ch:multiplane_dspl}: one null geodesic, several deflectors]
A compound lens is one null geodesic that passes two (or more) mass
concentrations at different redshifts.  Everything in Part~V rests on the
statement of this section (light follows null geodesics) plus one
weak-field fact from Chapter~\ref{ch:linearized_gravity}: a point mass bends the ray
almost entirely near closest approach (71\% of $\hat\alpha$ accrues within
$|x| < b$, 99.5\% within $|x| < 10b$).  The geodesic can therefore be
replaced by straight segments with a kink at each deflector, and that
replacement is the multi-plane lens equation.
\end{bridge}

\begin{bridge}[Chapter~\ref{ch:odd_images}: images are stationary null geodesics]
In a static weak field, the null geodesics between fixed endpoints are
exactly the paths that make the arrival time stationary (Fermat's
principle; eq.~\eqref{eq:fermat_principle} of
Chapter~\ref{ch:linearized_gravity} and Chapter~\ref{ch:fermat_time_delays}).  The geodesic picture of this chapter
and the arrival-time-surface picture used in Chapter~\ref{ch:odd_images} are
the same statement: each image is a null geodesic from source to observer,
and its type (minimum, saddle, maximum) is the type of the stationary
point.  The faint central image of a smooth lens is the maximum, and a
central point mass changes the count of such central images.
\end{bridge}

\section{Brief Overview: The Kerr Metric}
\label{sec:kerr_overview}

Real astrophysical black holes are not described exactly by the
Schwarzschild solution because they rotate.  The metric for a
rotating, axially symmetric black hole is the \emph{Kerr metric},
discovered by Roy Kerr in 1963.  In Boyer--Lindquist coordinates it
takes the form
\begin{equation}
    ds^2 = -\left(1 - \frac{R_S r}{\Sigma}\right) c^2 dt^2
    - \frac{2R_S r a \sin^2\theta}{\Sigma}\, c\, dt\, d\phi
    + \frac{\Sigma}{\Delta}\, dr^2
    + \Sigma\, d\theta^2
    + \frac{A \sin^2\theta}{\Sigma}\, d\phi^2 \,,
    \label{eq:kerr_metric}
\end{equation}
where
\begin{align}
    \Sigma &= r^2 + a^2\cos^2\theta \,, &
    \Delta &= r^2 - R_S r + a^2 \,, &
    A &= (r^2 + a^2)^2 - a^2 \Delta \sin^2\theta \,,
    \label{eq:kerr_functions}
\end{align}
and $a = J/(Mc)$ is the \emph{spin parameter} (angular momentum per
unit mass, with dimensions of length).  For a physical black hole,
$0 \leq a \leq GM/c^2$.

Four features distinguish the Kerr geometry from Schwarzschild.
The first is \emph{frame dragging}: the $dt\,d\phi$ cross-term forces
particles near the black hole to co-rotate, an effect
absent in Schwarzschild.
The second is the \emph{ergosphere}, the region
$r_+ < r < (R_S + \sqrt{R_S^2 - 4a^2\cos^2\theta})/2$,
where $r_+ = (R_S + \sqrt{R_S^2 - 4a^2})/2$ is the outer
event horizon, in which even stationary observers must co-rotate
with the black hole.  (At $a = 0$, $r_+ = R_S$, not
$R_S/2$, and the ergosphere collapses onto the Schwarzschild
horizon; only at maximum spin $a = GM/c^2$ does $r_+ = R_S/2$.)
The third is a spin-dependent ISCO: the ISCO radius depends on the spin
parameter and on the direction of the orbit (prograde or retrograde).
For a maximally spinning black hole ($a = GM/c^2$),
$r_{\mathrm{ISCO}} = GM/c^2$ (prograde) and
$r_{\mathrm{ISCO}} = 9GM/c^2$ (retrograde), compared with
$6GM/c^2$ for Schwarzschild.
The fourth is that there are two horizons,
$r_\pm = GM/c^2 \pm \sqrt{(GM/c^2)^2 - a^2}$.

\begin{annotation}[Checking the Kerr numbers]
The horizons are where $\Delta = 0 \iff
r = \tfrac12\bigl(R_S \pm \sqrt{R_S^2 - 4a^2}\bigr)
= GM/c^2 \pm \sqrt{(GM/c^2)^2 - a^2}$.  The ``two horizons''
expression and the $r_+$ of the ergosphere are the same
expression; both are real only for $a \le GM/c^2$.
The ergosurface is where $g_{tt} = 0 \iff \Sigma = R_Sr \iff
r^2 - R_Sr + a^2\cos^2\theta = 0$, with outer root
$\tfrac12\bigl(R_S + \sqrt{R_S^2 - 4a^2\cos^2\theta}\bigr)$.  It
touches the horizon at the poles ($\cos^2\theta = 1$) and reaches
$r = R_S$ at the equator for every $a$.
For the ISCO values ($G = c = M = 1$, $a$ in units of
$GM/c^2$), the standard closed form (Bardeen, Press \&
Teukolsky 1972)
$r_{\mathrm{ISCO}} = 3 + Z_2 \mp \sqrt{(3 - Z_1)(3 + Z_1 + 2Z_2)}$,
with $Z_1 = 1 + (1 - a^2)^{1/3}\bigl[(1 + a)^{1/3} + (1 - a)^{1/3}\bigr]$,
$Z_2 = \sqrt{3a^2 + Z_1^2}$ and the upper sign for prograde orbits,
gives $6$ at $a = 0$ and $1$ (prograde) or $9$ (retrograde) at
$a = 1$.
A unit trap: $a = J/(Mc)$ is a length, whereas papers usually
quote the dimensionless spin $a_* = ac^2/(GM) \in [0, 1]$.
\end{annotation}


\begin{remark}
For gravitational lensing by galaxies and clusters, the Kerr metric is
not needed: the lensing effect is dominated by the total enclosed mass,
and the weak-field/thin-lens approximation (Chapter~\ref{ch:linearized_gravity} and beyond) washes
out spin effects at the angular scales of interest.  Kerr lensing
is, however, an active research topic for strong lensing near individual
black holes, for instance the ``photon ring'' structure observed by
the Event Horizon Telescope, where frame-dragging produces asymmetric
brightness patterns.
\end{remark}

\section{Summary}
\label{sec:geodesics_summary}

The Killing vectors $\partial_t$ and $\partial_\phi$ yield a conserved
energy $E$ and angular momentum $L$ along geodesics, and the radial
motion reduces to a 1D effective potential problem with
$V_{\mathrm{eff}} = -\epsilon GM/r + L^2/(2r^2) - GML^2/(c^2 r^3)$.
The GR correction term $\propto 1/r^3$ is responsible for the
ISCO at $r = 3R_S$, the photon sphere at $r = 3R_S/2$, perihelion
precession, and enhanced light deflection.
Mercury's perihelion precesses by 43 arcsec/century, which is confirmed
by observation and was one of the first tests of GR.
The Kerr metric generalizes Schwarzschild to rotating black holes
(frame dragging, ergosphere, spin-dependent ISCO), but is not
needed for galaxy- or cluster-scale lensing.

\section{Problems}
\label{sec:geodesics_problems}

\begin{exercise}
\label{ex:derive_veff}
Starting from the normalization condition~\eqref{eq:normalization} and the
conserved quantities $E$ and $L$, derive the effective
potential~\eqref{eq:effective_potential} for massive particles
($\epsilon = 1$).  Show that in the Newtonian limit ($c \to \infty$,
$R_S \to 0$), it reduces to
$V_{\mathrm{eff}}^{\mathrm{Newton}} = -GM/r + L^2/(2r^2)$.
\end{exercise}

\begin{exercise}
\label{ex:isco_derivation}
Derive the ISCO radius by finding the value of $L$ for which the two
circular orbit solutions~\eqref{eq:circular_radii} coincide (i.e., the
discriminant vanishes).  Verify $r_{\mathrm{ISCO}} = 6GM/c^2 = 3R_S$.
\end{exercise}

\begin{exercise}
\label{ex:photon_sphere_derivation}
Show that the only circular photon orbit in the Schwarzschild spacetime is
at $r = 3GM/c^2 = 3R_S/2$, and that it is unstable (a maximum of
$V_{\mathrm{eff}}^{\mathrm{photon}}$).
\end{exercise}

\begin{exercise}
\label{ex:mercury_precession}
Compute the perihelion precession of Mercury using
eq.~\eqref{eq:perihelion_precession} with:
$M = M_\odot$, $a = 5.79 \times 10^{10}$~m, $e = 0.2056$.
Verify $\Delta\phi \approx 43''$/century (Mercury's orbital period is
87.97 days).
\end{exercise}

\begin{exercise}
\label{ex:orbit_integration}
Using the effective potential for massive particles with $L = 4GM/c$:
\begin{enumerate}[label=(\alph*)]
    \item Plot $V_{\mathrm{eff}}(r)$ and identify the stable circular
          orbit radius, the unstable circular orbit radius, and the
          ISCO.
    \item For a particle with $\mathcal{E}$ slightly above
          $V_{\mathrm{eff}}(r_+)$, describe qualitatively the type of
          orbit (bound precessing ellipse, scatter orbit, or plunge orbit).
\end{enumerate}
\end{exercise}
