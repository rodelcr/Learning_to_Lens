\subsection*{Exercise~\ref{ex:derive_veff}: Deriving the Effective Potential}

We start from the normalization condition~\eqref{eq:normalization} with
$f(r) = 1 - R_S/r$:
\begin{equation*}
    -f c^2\!\left(\frac{dt}{d\lambda}\right)^{\!2}
    + \frac{1}{f}\left(\frac{dr}{d\lambda}\right)^{\!2}
    + r^2\!\left(\frac{d\phi}{d\lambda}\right)^{\!2} = -\epsilon c^2 \,.
\end{equation*}
Substituting the conserved quantities $E = f c^2\, dt/d\lambda$ and
$L = r^2\, d\phi/d\lambda$:
\begin{equation*}
    -\frac{E^2}{f c^2} + \frac{1}{f}\left(\frac{dr}{d\lambda}\right)^{\!2}
    + \frac{L^2}{r^2} = -\epsilon c^2 \,.
\end{equation*}
Multiplying through by $f$ and rearranging:
\begin{align*}
    \left(\frac{dr}{d\lambda}\right)^{\!2}
    &= \frac{E^2}{c^2} - f(r)\!\left(\epsilon c^2 + \frac{L^2}{r^2}\right) \\
    &= \frac{E^2}{c^2} - \left(1 - \frac{R_S}{r}\right)\!\left(\epsilon c^2 + \frac{L^2}{r^2}\right) \\
    &= \frac{E^2}{c^2} - \epsilon c^2 + \frac{\epsilon c^2 R_S}{r} - \frac{L^2}{r^2} + \frac{R_S L^2}{r^3} \,.
\end{align*}
Dividing by 2 and identifying terms:
\begin{equation*}
    \frac{1}{2}\left(\frac{dr}{d\lambda}\right)^{\!2}
    + \underbrace{\left(-\frac{\epsilon GM}{r}
    + \frac{L^2}{2r^2}
    - \frac{GML^2}{c^2 r^3}\right)}_{V_{\mathrm{eff}}(r)}
    = \frac{1}{2}\left(\frac{E^2}{c^2} - \epsilon c^2\right) \equiv \mathcal{E} \,,
\end{equation*}
where we used $R_S = 2GM/c^2$.  This gives the effective
potential~\eqref{eq:effective_potential}:
\[
    \boxed{V_{\mathrm{eff}}(r) = -\frac{\epsilon\, GM}{r}
    + \frac{L^2}{2r^2} - \frac{GML^2}{c^2 r^3}}
\]

\begin{derivation}[The ``identifying terms'' step, with $c$ kept]
Halving the last expanded line term by term with $R_S = 2GM/c^2$
(see also the derivation box after eq.~\eqref{eq:effective_potential}):
\begin{align*}
\tfrac12\,\epsilon c^2R_S/r &= \epsilon GM/r
  && \text{(moves left as } -\epsilon GM/r\text{)} \\
\tfrac12\,L^2/r^2 &= L^2/(2r^2)
  && \text{(moves left as } +L^2/(2r^2)\text{)} \\
\tfrac12\,R_SL^2/r^3 &= GML^2/(c^2r^3)
  && \text{(moves left as } -GML^2/(c^2r^3)\text{)} .
\end{align*}
The remaining two terms, $\tfrac12(E^2/c^2 - \epsilon c^2)$, are the
constant $\mathcal{E}$.  With $c = 1$ (Carroll's units) every $c$ above
drops out and $\mathcal{E} = \tfrac12(E^2 - \epsilon)$.
\end{derivation}


The last term $-GML^2/(c^2 r^3)$ is the GR
correction.  In the Newtonian limit ($c \to \infty$, or equivalently
$R_S \to 0$), this term vanishes, leaving
\[
    V_{\mathrm{eff}}^{\mathrm{Newton}} = -\frac{GM}{r} + \frac{L^2}{2r^2} \,,
\]
which is the standard Newtonian effective potential.  The centrifugal
barrier $L^2/(2r^2)$ diverges as $r \to 0$, preventing any particle from
reaching the origin.  The GR correction term is attractive at small $r$
and allows the barrier to be overcome, which is why particles can plunge
into black holes.

\begin{annotation}[What ``$c \to \infty$'' means]
The GR term is smaller than the centrifugal term by the factor
$\bigl(GML^2/c^2r^3\bigr)/\bigl(L^2/2r^2\bigr) = R_S/r$.  Sending
$c \to \infty$ at fixed $r$ is a shorthand for the physical condition
$r \gg R_S$.  For the same reason $\mathcal{E} = \tfrac12(E^2/c^2 - c^2)$ tends
to the Newtonian specific energy: with $E = c^2 + e$,
$\mathcal{E} = e + e^2/(2c^2)$.
\end{annotation}

\subsection*{Exercise~\ref{ex:isco_derivation}: ISCO Derivation}

Working in $G = c = M = 1$ units, the effective potential for massive
particles is $V_{\mathrm{eff}} = -1/r + L^2/(2r^2) - L^2/r^3$.

Step 1 is the circular orbit condition, $dV_{\mathrm{eff}}/dr = 0$:
\begin{equation*}
    \frac{1}{r^2} - \frac{L^2}{r^3} + \frac{3L^2}{r^4} = 0
    \quad\Longrightarrow\quad
    r^2 - L^2 r + 3L^2 = 0 \,.
\end{equation*}
The two solutions are
\begin{equation*}
    r_\pm = \frac{L^2 \pm \sqrt{L^4 - 12L^2}}{2}
    = \frac{L^2}{2}\left(1 \pm \sqrt{1 - \frac{12}{L^2}}\right) \,.
\end{equation*}
Step 2 is the ISCO condition: the ISCO occurs when the two circular
orbits merge, i.e., when the discriminant vanishes,
\begin{equation*}
    L^4 - 12L^2 = 0 \quad\Longrightarrow\quad L^2(L^2 - 12) = 0
    \quad\Longrightarrow\quad L = 2\sqrt{3} \,.
\end{equation*}
Step 3 gives the ISCO radius; substituting $L^2 = 12$,
\begin{equation*}
    r_{\mathrm{ISCO}} = \frac{12}{2} = \boxed{6} \quad
    \left(= \frac{6GM}{c^2} = 3\,R_S\right) \,.
\end{equation*}
As a verification via $d^2V/dr^2 = 0$, we can alternatively solve
$dV/dr = 0$ and $d^2V/dr^2 = 0$ simultaneously, with
\begin{equation*}
    \frac{d^2 V_{\mathrm{eff}}}{dr^2} = -\frac{2}{r^3} + \frac{3L^2}{r^4}
    - \frac{12L^2}{r^5} = 0 \,.
\end{equation*}
Combined with $dV/dr = 0$, this yields $r = 6$, $L = 2\sqrt{3}$,
confirming the result.

\begin{derivation}[Solving the two conditions simultaneously]
We multiply $dV/dr = 0$ by $r^4$ and $d^2V/dr^2 = 0$ by $r^5$:
\begin{align*}
r^2 - L^2r + 3L^2 &= 0 && \text{(i)} \\
-2r^2 + 3L^2r - 12L^2 &= 0 && \text{(ii)} \\
L^2r - 6L^2 = L^2(r - 6) &= 0 && \bigl(2\times\text{(i)} + \text{(ii)}\bigr) \\
r &= 6 && (L \neq 0) \\
36 - 6L^2 + 3L^2 = 36 - 3L^2 &= 0 \;\Rightarrow\; L^2 = 12
  && (\text{back into (i)}) .
\end{align*}
The condition $V' = V'' = 0$ marks an inflection point: the minimum at $r_+$
and the maximum at $r_-$ have merged, which is the same condition as the
vanishing discriminant of Step~2.
\end{derivation}

\subsection*{Exercise~\ref{ex:photon_sphere_derivation}: Photon Sphere}

For photons ($\epsilon = 0$), the effective potential is (in $G = c = M = 1$ units)
\begin{equation*}
    V_{\mathrm{eff}}^{\mathrm{photon}} = \frac{L^2}{2r^2} - \frac{L^2}{r^3} \,.
\end{equation*}
For the circular orbit, setting $dV/dr = 0$ gives
\begin{equation*}
    -\frac{L^2}{r^3} + \frac{3L^2}{r^4} = 0
    \quad\Longrightarrow\quad
    \frac{L^2}{r^4}(3 - r) = 0
    \quad\Longrightarrow\quad
    \boxed{r = 3}
    \quad\left(= \frac{3GM}{c^2} = \frac{3}{2}\,R_S\right) \,.
\end{equation*}

For the stability, we evaluate the second derivative at $r = 3$:
\begin{equation*}
    \frac{d^2 V_{\mathrm{eff}}}{dr^2}\bigg|_{r=3}
    = \left(\frac{3L^2}{r^4} - \frac{12L^2}{r^5}\right)\bigg|_{r=3}
    = \frac{3L^2}{81} - \frac{12L^2}{243}
    = \frac{L^2}{243}(9 - 12)
    = -\frac{L^2}{81} < 0 \,.
\end{equation*}
Since $d^2V/dr^2 < 0$, the effective potential has a maximum
at $r = 3$, confirming that the photon orbit is unstable.
Any perturbation causes the photon to spiral inward (capture) or
escape to infinity.

\begin{annotation}[Alternative route: the barrier height gives the capture impact parameter]
The maximum of the barrier is
$V^{\mathrm{photon}}_{\mathrm{eff}}(3) = L^2/18 - L^2/27 = L^2/54$.  A photon
from infinity has $\mathcal{E} = \tfrac12E^2$ (with $\epsilon = 0$ there is
no rest-mass term), so it is turned back iff $\tfrac12E^2 < L^2/54$, i.e.\
iff $b^2 = L^2/E^2 > 27$: $b_{\mathrm{crit}} = 3\sqrt3\,GM/c^2 \approx
2.598\,R_S$.  The radius $r = 3$ is independent of $L$; the photon orbit
has no analogue of the $L$-dependent pair $r_\pm$ of Exercise~\ref{ex:isco_derivation}.
\end{annotation}

\subsection*{Exercise~\ref{ex:mercury_precession}: Mercury's Perihelion Precession}

Using eq.~\eqref{eq:perihelion_precession} with Mercury's orbital parameters:
\begin{alignat*}{2}
    G &= 6.674 \times 10^{-11} \; \text{m}^3\,\text{kg}^{-1}\,\text{s}^{-2} & \qquad
    M_\odot &= 1.989 \times 10^{30} \; \text{kg} \\
    a &= 5.79 \times 10^{10} \; \text{m} & \qquad
    e &= 0.2056
\end{alignat*}

Step 1 is the precession per orbit:
\begin{align*}
    \Delta\phi &= \frac{6\pi G M}{c^2\, a\,(1 - e^2)} \\
    &= \frac{6\pi \times 6.674\!\times\!10^{-11} \times 1.989\!\times\!10^{30}}
    {(2.998\!\times\!10^{8})^2 \times 5.79\!\times\!10^{10} \times (1 - 0.2056^2)} \\
    &= 5.02 \times 10^{-7} \;\text{rad/orbit} \,.
\end{align*}

\begin{derivation}[The arithmetic in Step 1, organised by length scales]
\begin{align*}
\frac{GM_\odot}{c^2} &= \frac{6.674\times10^{-11}\times1.989\times10^{30}}
  {(2.998\times10^8)^2}~\text{m} = 1477~\text{m}
  && \text{(the Sun's gravitational radius)} \\
6\pi\,\frac{GM_\odot}{c^2} &= 2.784\times10^4~\text{m} \\
a(1 - e^2) &= 5.79\times10^{10}\times0.9577~\text{m} = 5.545\times10^{10}~\text{m}
  && \text{(semi-latus rectum)} \\
\Delta\phi &= \frac{2.784\times10^4}{5.545\times10^{10}}
  = 5.02\times10^{-7}~\text{rad/orbit}
  && \text{(dimensionless: m/m)} .
\end{align*}
Mathematica (\texttt{problems\_01d.wl}, annotated-edition checks) gives
$5.020\times10^{-7}$~rad $= 0.10355''$ per orbit.  Keep four significant
figures in Step~2 ($0.1036''$): then $0.1036\times415.2 = 43.0''$.  (Earlier
versions of Steps~1--2 printed $5.01\times10^{-7}$ and $0.1034''$, a slip;
$0.1034\times415.2 = 42.9''$.)  Step~3 uses the Julian year, 365.25~days.
\end{derivation}


Step 2 converts this to arcseconds:
\begin{equation*}
    \Delta\phi = 5.02 \times 10^{-7} \;\text{rad}
    \times \frac{180 \times 3600}{\pi} \;\frac{\text{arcsec}}{\text{rad}}
    = 0.1036\;\text{arcsec/orbit} \,.
\end{equation*}

Step 3 counts the orbits per century:
\begin{equation*}
    N_{\mathrm{orbits}} = \frac{100 \times 365.25}{87.97} = 415.2 \;\text{orbits/century} \,.
\end{equation*}

Step 4 gives the precession per century:
\begin{equation*}
    \Delta\phi_{\mathrm{century}}
    = 0.1036 \times 415.2
    = \boxed{43.0 \;\text{arcsec/century}} \,.
\end{equation*}
This matches the observed anomalous precession of Mercury, confirming one
of the three classical tests of general relativity.

\subsection*{Exercise~\ref{ex:orbit_integration}: Effective Potential for $L = 4$}

(a) Working in $G = c = M = 1$ units with $L = 4$, the circular
orbit radii are found by solving $dV_{\mathrm{eff}}/dr = 0$:
\begin{equation*}
    r^2 - 16r + 48 = 0
    \quad\Longrightarrow\quad
    r = \frac{16 \pm \sqrt{256 - 192}}{2}
    = \frac{16 \pm 8}{2} \,.
\end{equation*}
The stable circular orbit (minimum of $V_{\mathrm{eff}}$) is therefore at
$r_+ = 12$, where $V_{\mathrm{eff}}(12) = -1/12 + 16/288 - 16/1728 = -1/27 \approx -0.0370$,
and the unstable circular orbit (maximum of $V_{\mathrm{eff}}$) at
$r_- = 4$, where $V_{\mathrm{eff}}(4) = -1/4 + 16/32 - 16/64 = -0.25 + 0.5 - 0.25 = 0$.
For reference, the ISCO $r_{\mathrm{ISCO}} = 6$ occurs at
$L = 2\sqrt{3} \approx 3.464$, which is below $L = 4$, so the two
circular orbits are well separated.
The plot (see \solnlink{01d_Geodesics_Orbits/problems\_01d.wl}{problems\_01d.wl})
shows the GR potential (solid) with the Newtonian potential (dashed) for
comparison.  The GR correction causes $V_{\mathrm{eff}} \to -\infty$ as
$r \to 0$, allowing plunge orbits.

(b) For $\mathcal{E}$ slightly above $V_{\mathrm{eff}}(r_+)$, the
particle oscillates between a minimum radius (perihelion) and a maximum
radius (aphelion), but the orbit does not close due to the GR correction
term.  The perihelion advances by a large angle each orbit
($\approx 2.6$~rad $\approx 149^\circ$ for near-circular orbits at
$r \approx 12\,GM/c^2$), producing a strongly precessing rosette, a
bound precessing ellipse in the classification of the exercise.
Only for $r \gg 6GM/c^2$ does the advance become small, tending to
$6\pi GM/(c^2 r)$ per orbit: the regime of Mercury's perihelion precession
(Exercise~\ref{ex:mercury_precession}).

\begin{annotation}[Why $V_{\mathrm{eff}}(4) = 0$ is special]
Since $\mathcal{E} = \tfrac12(E^2/c^2 - c^2) = \tfrac12(E^2 - 1)$ in these units, the unstable orbit at
$r_- = 4$ has $E^2 = 1 + 2V_{\mathrm{eff}}(4) = 1$: it is \emph{marginally
bound}.  A particle released from rest at infinity with $L = 4\,GM/c$ just
reaches the top of the barrier at $r = 4GM/c^2$.
\end{annotation}

\begin{derivation}[How big is the precession for $L = 4$? (part b, quantitatively)]
For a nearly circular orbit at $r_c$, we linearise $\ddot r = -V_{\mathrm{eff}}'(r)$
about $r_c$ to obtain $\delta\ddot r = -V_{\mathrm{eff}}''(r_c)\,\delta r$, a radial
oscillation with $\kappa^2 = V_{\mathrm{eff}}''(r_c)$ (per unit proper time).
The angular frequency is $\Omega = d\phi/d\tau = L/r_c^2$.  On a circular
orbit $r_c^2 - L^2r_c + 3L^2 = 0$, i.e.\ $L^2 = r_c^2/(r_c - 3)$, and with
$V_{\mathrm{eff}}'' = -2/r^3 + 3L^2/r^4 - 12L^2/r^5$:
\[
\frac{\kappa^2}{\Omega^2} = 1 - \frac{6}{r_c}
\qquad (G = c = M = 1).
\]
Per radial period, $\phi$ advances by $2\pi\,\Omega/\kappa = 2\pi/\sqrt{1 - 6/r_c}$.
At $r_c = 12$ (which indeed has $L^2 = 144/9 = 16$) the excess over $2\pi$ is
\[
2\pi\bigl(\sqrt2 - 1\bigr) \approx 2.60~\text{rad} \approx 149^\circ
\quad\text{per orbit},
\]
as quoted in part~(b).  For
$r_c \gg 6$, $2\pi\bigl(1/\sqrt{1 - 6/r_c} - 1\bigr) \to 6\pi/r_c =
6\pi GM/(c^2r_c)$, the $e \to 0$ limit of eq.~\eqref{eq:perihelion_precession}.
\end{derivation}
